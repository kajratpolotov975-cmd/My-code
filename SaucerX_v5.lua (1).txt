-- шарнир вокруг оси X локальной системы H (для капота и багажника)
local function hinge(group, weld, closed, H, ang, part, collFn)
	local Hi = H:Inverse()
	table.insert(anim[group], {
		w = weld,
		part = part,
		coll = collFn,
		fn = function(a) return H * CFrame.Angles(ang * a, 0, 0) * Hi * closed end,
	})
end

-- шарнир вокруг точки H на дуге купола (для дверей, которые открываются вбок наружу и чуть вниз как на фото)
local function hingeDoor(group, weld, closed, H, openYaw, openPitch, part, collFn)
	local Hi = H:Inverse()
	table.insert(anim[group], {
		w = weld,
		part = part,
		coll = collFn,
		fn = function(a)
			return H * CFrame.Angles(0, openYaw * a, 0) * CFrame.Angles(openPitch * a, 0, 0) * Hi * closed
		end,
	})
end

local function applyAlpha(group, a)
	if group == "gear" then
		updateWheels()
		return
	end
	for _, e in ipairs(anim[group]) do
		if e.w and e.w.Parent then
			e.w.C0 = e.fn(a)
			if e.part and e.coll then e.part.CanCollide = e.coll(a) end
		end
	end
end

local function setGroup(group, open, instant)
	state[group] = open
	gen[group] = gen[group] + 1
	local my = gen[group]
	local from = alpha[group]
	local to = open and 1 or 0
	if instant or from == to then
		alpha[group] = to
		applyAlpha(group, to)
	else
		task.spawn(function()
			local t0 = os.clock()
			local dur = (group == "gear" and 0.9 or 1.05) * math.abs(to - from)
			while gen[group] == my do
				local p = math.min((os.clock() - t0) / dur, 1)
				local e = (p < 0.5) and (2 * p * p) or (1 - 2 * (1 - p) * (1 - p))
				alpha[group] = from + (to - from) * e
				applyAlpha(group, alpha[group])
				if p >= 1 then break end
				RunService.Heartbeat:Wait()
			end
		end)
	end
	if refreshUI then refreshUI() end
end

local BLOCK, BALL, CYL = Enum.PartType.Block, Enum.PartType.Ball, Enum.PartType.Cylinder
local MAT = Enum.Material

local N       = 16                       -- граней по кругу
local STEP    = 360 / N                  -- 22.5 градуса
local TANH    = math.tan(math.rad(STEP / 2))
local COSH    = math.cos(math.rad(STEP / 2))
local Y0, Y1, Y2 = 0.1, 0.95, 2.4        -- высота ярусов борта
local R0, R2  = 10.6, 7.6                -- ШИРЕ: радиус низа 10.6 (диаметр 21.2), верха 7.6
local DR, DH  = 5.8, 4.8                 -- БОЛЬШЕ МЕСТА: радиус стеклянной капсулы 5.8 (диаметр 11.6), высота 4.8
local HINGE_R = 5.85                     -- шарнир капота и багажника у обода купола

-- номера граней (j = 0..15), центр грани = 11.25 + 22.5 * j градусов
local DOOR_J  = { [3] = true, [4] = true, [11] = true, [12] = true }
local HOOD_J  = { [15] = true, [0] = true }
local TRUNK_J = { [7] = true, [8] = true }

local function rOf(y) return R0 - (y - Y0) * (R0 - R2) / (Y2 - Y0) end
local function thetaOf(j) return math.rad(STEP / 2 + STEP * j) end
local function wallW(r) return 2 * r * TANH + 0.08 end

-- плоская панель на конусе (или куполе) между двумя кольцами
local function slabCF(theta, ya, ra, yb, rb)
	local dy, dr = yb - ya, ra - rb
	local len = math.sqrt(dy * dy + dr * dr)
	local a = math.atan2(dr, dy)
	local cf = CFrame.Angles(0, theta, 0) * CFrame.new(0, (ya + yb) / 2, -(ra + rb) / 2) * CFrame.Angles(a, 0, 0)
	return cf, len
end

local function mk(name, shape, size, rel, color, mat, o)
	o = o or {}
	local p = Instance.new("Part")
	p.Name = name
	p.Shape = shape
	p.Size = size
	p.Color = color
	p.Material = mat or MAT.Metal
	p.Anchored = false
	p.CanCollide = o.collide or false
	p.CanTouch = false
	p.CanQuery = o.query ~= false
	p.Massless = true
	p.Transparency = o.tr or 0
	p.Reflectance = o.refl or 0
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.CFrame = root.CFrame * rel
	p.Parent = ship
	local w = Instance.new("Weld")
	w.Part0 = root
	w.Part1 = p
	w.C0 = rel
	w.Parent = p
	return p, w
end

local function disc(name, y, h, d, color, mat, o)
	return mk(name, CYL, Vector3.new(h, d, d), CFrame.new(0, y, 0) * CFrame.Angles(0, 0, math.pi / 2), color, mat, o)
end

-- цилиндр вдоль оси Z (вперёд-назад)
local function zcyl(name, len, d, rel, color, mat, o)
	return mk(name, CYL, Vector3.new(len, d, d), rel * CFrame.Angles(0, math.pi / 2, 0), color, mat, o)
end

-- вертикальный цилиндр
local function vcyl(name, h, d, rel, color, mat, o)
	return mk(name, CYL, Vector3.new(h, d, d), rel * CFrame.Angles(0, 0, math.pi / 2), color, mat, o)
end

local function addLight(part, color, brightness, range)
	local l = Instance.new("PointLight")
	l.Color = color; l.Brightness = brightness; l.Range = range
	l.Parent = part
	return l
end

-- деталь, которая вращается вокруг оси (вентилятор, стрелки, руль)
local function rotAxis(axis, a)
	if axis == "X" then return CFrame.Angles(a, 0, 0) end
	if axis == "Y" then return CFrame.Angles(0, a, 0) end
	return CFrame.Angles(0, 0, a)
end

local function spinPart(name, shape, size, frame, off, color, mat, axis, getter, o)
	local p, w = mk(name, shape, size, frame * off, color, mat, o)
	table.insert(spinners, { w = w, cf = frame, off = off, axis = axis, get = getter })
	return p
end

local function addPulse(part, light, base, kind)
	table.insert(pulsers, { part = part, light = light, base = base, kind = kind or 0 })
end

local function updateSpinners()
	for _, e in ipairs(spinners) do
		if e.w and e.w.Parent then
			e.w.C0 = e.cf * rotAxis(e.axis, e.get()) * e.off
		end
	end
end

local function updatePulses(t)
	for i, e in ipairs(pulsers) do
		local k = 0.5 + 0.5 * math.sin(t * 3 + i * 1.7)
		if e.part and e.part.Parent then e.part.Transparency = 0.05 + 0.25 * (1 - k) end
		if e.light and e.light.Parent then e.light.Brightness = e.base * (0.6 + 0.6 * k) end
	end
end

-- Анимация синего огня из боковых капсул при полёте

local function updatePodFlames(t, dt, active, boostMult, speedRatio)
	local target = active and math.clamp(0.55 + speedRatio * 0.45 + (boostMult > 1 and 0.35 or 0), 0.55, 1.35) or 0
	flamePower = flamePower + (target - flamePower) * math.min(dt * 10, 1)
	local show = flamePower > 0.03

	for idx, f in ipairs(podFlames) do
		if f.emitter and f.emitter.Parent then
			f.emitter.Enabled = show
			if show then
				f.emitter.Rate = math.floor(65 * flamePower)
				local spd = 16 + 22 * flamePower
				f.emitter.Speed = NumberRange.new(spd * 0.75, spd * 1.15)
			end
		end

		if not show then
			if f.outer and f.outer.Parent then f.outer.Transparency = 1 end
			if f.mid and f.mid.Parent then f.mid.Transparency = 1 end
			if f.core and f.core.Parent then f.core.Transparency = 1 end
			if f.tip and f.tip.Parent then f.tip.Transparency = 1 end
			if f.ring1 and f.ring1.Parent then f.ring1.Transparency = 1 end
			if f.ring2 and f.ring2.Parent then f.ring2.Transparency = 1 end
			if f.light and f.light.Parent then f.light.Brightness = 1.8 end
		else
			local flick1 = 0.88 + 0.22 * math.sin(t * 42 + idx * 2.1) + 0.1 * math.cos(t * 73 + idx)
			local flick2 = 0.90 + 0.18 * math.cos(t * 55 + idx * 1.4)
			local lenOuter = (3.4 + 2.8 * (flamePower - 0.5)) * flick1
			local lenMid   = lenOuter * 0.76 * flick2
			local lenCore  = lenOuter * 0.48

			if f.outer and f.outerW and f.outer.Parent then
				f.outer.Size = Vector3.new(lenOuter, 2.05 * flick2, 2.05 * flick2)
				f.outerW.C0 = f.baseCF * CFrame.new(0, 0, 2.32 + lenOuter / 2) * CFrame.Angles(0, math.pi / 2, 0)
				f.outer.Transparency = math.clamp(0.32 - 0.12 * flamePower + 0.08 * math.sin(t * 38), 0.15, 0.55)
			end
			if f.mid and f.midW and f.mid.Parent then
				f.mid.Size = Vector3.new(lenMid, 1.42 * flick1, 1.42 * flick1)
				f.midW.C0 = f.baseCF * CFrame.new(0, 0, 2.30 + lenMid / 2) * CFrame.Angles(0, math.pi / 2, 0)
				f.mid.Transparency = math.clamp(0.18 + 0.08 * math.cos(t * 48), 0.1, 0.35)
			end
			if f.core and f.coreW and f.core.Parent then
				f.core.Size = Vector3.new(lenCore, 0.86, 0.86)
				f.coreW.C0 = f.baseCF * CFrame.new(0, 0, 2.28 + lenCore / 2) * CFrame.Angles(0, math.pi / 2, 0)
				f.core.Transparency = 0.05
			end
			if f.tip and f.tipW and f.tip.Parent then
				local td = 1.25 * flick1
				f.tip.Size = Vector3.new(td, td, td * 1.4)
				f.tipW.C0 = f.baseCF * CFrame.new(0, 0, 2.32 + lenOuter)
				f.tip.Transparency = 0.38
			end

			-- бегущие плазменные кольца вдоль струи синего огня
			local p1 = (t * 4.5 + idx * 0.3) % 1
			local p2 = (t * 4.5 + idx * 0.3 + 0.5) % 1
			if f.ring1 and f.ring1W and f.ring1.Parent then
				local d1 = 2.3 * (1 - p1 * 0.55)
				f.ring1.Size = Vector3.new(0.14, d1, d1)
				f.ring1W.C0 = f.baseCF * CFrame.new(0, 0, 2.35 + p1 * lenOuter * 0.95) * CFrame.Angles(0, math.pi / 2, 0)
				f.ring1.Transparency = 0.15 + p1 * 0.75
			end
			if f.ring2 and f.ring2W and f.ring2.Parent then
				local d2 = 2.3 * (1 - p2 * 0.55)
				f.ring2.Size = Vector3.new(0.14, d2, d2)
				f.ring2W.C0 = f.baseCF * CFrame.new(0, 0, 2.35 + p2 * lenOuter * 0.95) * CFrame.Angles(0, math.pi / 2, 0)
				f.ring2.Transparency = 0.15 + p2 * 0.75
			end

			if f.light and f.light.Parent then
				f.light.Brightness = 2.6 + 2.2 * flamePower * flick1
				f.light.Range = 22 + 10 * flamePower
			end
		end
	end
end

-- крышка: капот (dir = -1, спереди) или багажник (dir = +1, сзади).
local function compartment(group, dir)
	local axisTheta = (dir < 0) and 0 or math.pi
	local js = (dir < 0) and { 15, 0 } or { 7, 8 }
	local H = CFrame.Angles(0, axisTheta, 0) * CFrame.new(0, Y2, -HINGE_R)
	local LID_ANGLE = math.rad(92)

	local function add(name, shape, size, rel, color, mat, o)
		local p, w = mk(name, shape, size, rel, color, mat, o)
		hinge(group, w, rel, H, LID_ANGLE)
		return p
	end

	local flatMid = (HINGE_R + R2) / 2
	for idx, j in ipairs(js) do
		local th = thetaOf(j)
		add(group .. "LidTop", BLOCK, Vector3.new(wallW(R2), 0.3, R2 - HINGE_R),
			CFrame.Angles(0, th, 0) * CFrame.new(0, Y2 - 0.15 + (idx - 1) * 0.012, -flatMid), C.hull, MAT.Metal)
		local scf, slen = slabCF(th, Y1, rOf(Y1), Y2, R2)
		add(group .. "LidSlope", BLOCK, Vector3.new(wallW(rOf(Y1)), slen, 0.3), scf, C.hull, MAT.Metal)
	end

	-- ЖЁЛТАЯ ПОЛОСА по центру капота и багажника (как на кадре)
	add(group .. "LidStripe", BLOCK, Vector3.new(0.32, 0.05, R2 - HINGE_R - 0.25),
		CFrame.Angles(0, axisTheta, 0) * CFrame.new(0, Y2 + 0.03, -flatMid), C.orange, MAT.Neon)
	local lcf, llen = slabCF(axisTheta, Y1, rOf(Y1), Y2, R2)
	add(group .. "LidSlopeStripe", BLOCK, Vector3.new(0.32, llen, 0.06), lcf * CFrame.new(0, 0, -0.17), C.orange, MAT.Neon)
	for i = 1, 5 do
		add(group .. "LidLED", BLOCK, Vector3.new(0.38, 0.05, 0.08), lcf * CFrame.new(0, -llen / 2 + (i - 0.5) * llen / 5, -0.2), C.ledDark, MAT.Metal)
	end
	local wcf, wlen = slabCF(axisTheta, 0.25, rOf(0.25), Y1, rOf(Y1))
	mk(group .. "BodyStripe", BLOCK, Vector3.new(0.32, wlen, 0.06), wcf * CFrame.new(0, 0, -0.17), C.orange, MAT.Neon)
	for i = 1, 4 do
		mk(group .. "BodyLED", BLOCK, Vector3.new(0.38, 0.05, 0.08), wcf * CFrame.new(0, -wlen / 2 + (i - 0.5) * wlen / 4, -0.2), C.ledDark, MAT.Metal)
	end

	local hcf = slabCF(axisTheta, 1.1, rOf(1.1), 1.5, rOf(1.5)) * CFrame.new(0, 0, -0.2)
	add(group .. "LidHandle", BLOCK, Vector3.new(1.6, 0.16, 0.16), hcf, C.dark, MAT.Metal)

	if dir < 0 then
		-- передняя верхняя плита с двумя круглыми винтами (как на фото спереди купола)
		local plateCF = CFrame.Angles(0, axisTheta, 0) * CFrame.new(0, Y2 + 0.12, -(HINGE_R + 0.75))
		add("LidPlate", BLOCK, Vector3.new(5.4, 0.36, 1.4), plateCF, C.hull, MAT.Metal)
		for _, sx in ipairs({ -1.5, 1.5 }) do
			local sc = plateCF * CFrame.new(sx, 0.02, -0.68)
			add("Screw", CYL, Vector3.new(0.12, 0.44, 0.44), sc * CFrame.Angles(0, math.pi / 2, 0), C.hull2, MAT.Metal)
			add("ScrewSlot", BLOCK, Vector3.new(0.04, 0.34, 0.06), sc * CFrame.new(0, 0, -0.05) * CFrame.Angles(0, 0, 0.25), C.dark, MAT.Metal)
		end
	end

	-- перегородка сдвинута дальше (5.4 стада), чтобы в салоне было много места
	mk(group .. "Firewall", BLOCK, Vector3.new(4.2, 1.8, 0.2), CFrame.new(0, 1.4, dir * 5.4), C.hull2)
	local L = CFrame.new(0, 0.5, dir * 7.0) * ((dir > 0) and CFrame.Angles(0, math.pi, 0) or CFrame.new())
	if dir < 0 then hoodBay(L) else trunkBay(L) end
end

-- банки и бутылки в салоне и у порога двери (как на фото, но CanCollide = false — ходить не мешают)
local function clutter()
	local rnd = Random.new(11)
	for _, sx in ipairs({ -1, 1 }) do
		for i = 1, 12 do
			local x = sx * (3.1 + rnd:NextNumber() * 3.8)
			local z = -2.6 + rnd:NextNumber() * 5.2
			local yw = rnd:NextNumber() * math.pi * 2
			if i % 3 == 0 then
				local col = (i % 2 == 0) and C.bottle or C.bottle2
				local base = CFrame.new(x, 0.72, z) * CFrame.Angles(0, yw, math.pi / 2)
				mk("Bottle", CYL, Vector3.new(1.15, 0.44, 0.44), base, col, MAT.Glass, { tr = 0.12, query = false })

			else
				local cf = CFrame.new(x, 0.72, z) * CFrame.Angles(0, yw, 0)
				mk("Can", CYL, Vector3.new(0.75, 0.46, 0.46), cf, C.can, MAT.Metal, { query = false })
				mk("CanCap", CYL, Vector3.new(0.1, 0.48, 0.48), cf * CFrame.new(0.33, 0, 0), C.silver, MAT.Metal, { query = false })
			end
		end
	end
end

local function buildWheels()
	for _, sx in ipairs({ -6.2, 6.2 }) do
		for _, sz in ipairs({ -4.2, 4.2 }) do
			local out = CFrame.new(sx, -0.9, sz)
			local inn = CFrame.new(sx * 0.78, -0.1, sz) * CFrame.Angles(0, 0, math.pi / 2)
			local face = (sx < 0) and -1 or 1
			local wh = { out = out, inn = inn, parts = {} }
			local function part(name, shape, size, off, color, mat, spin)
				local _, w = mk(name, shape, size, out * off, color, mat)
				table.insert(wh.parts, { w = w, off = off, spin = spin })
			end
			part("Tire", CYL, Vector3.new(0.95, 2.25, 2.25), CFrame.new(), C.black, MAT.Slate, true)
			part("Rim", CYL, Vector3.new(1.0, 1.35, 1.35), CFrame.new(), C.hull2, MAT.Metal, true)
			part("HubCap", CYL, Vector3.new(1.05, 0.48, 0.48), CFrame.new(), C.silver, MAT.Metal, true)
			for i = 0, 7 do
				part("Tread", BLOCK, Vector3.new(0.96, 0.16, 0.5), CFrame.Angles(i * math.pi / 4, 0, 0) * CFrame.new(0, 1.12, 0), C.black, MAT.Slate, true)
			end
			for i = 0, 4 do
				part("Lug", CYL, Vector3.new(0.1, 0.17, 0.17), CFrame.Angles(i * math.pi * 2 / 5, 0, 0) * CFrame.new(face * 0.54, 0.42, 0), C.silver, MAT.Metal, true)
			end
			part("Strut", BLOCK, Vector3.new(0.5, 1.7, 0.5), CFrame.new(face * 0.1, 1.0, 0), C.dark, MAT.Metal, false)
			part("Piston", CYL, Vector3.new(1.2, 0.25, 0.25), CFrame.new(face * 0.4, 1.2, 0) * CFrame.Angles(0, 0, math.pi / 2), C.silver, MAT.Metal, false)
			table.insert(wheels, wh)
		end
	end
end

function updateWheels()
	local a = alpha.gear
	if a == lastWheelA and wheelSpin == lastWheelSpin then return end
	lastWheelA, lastWheelSpin = a, wheelSpin
	local sp = CFrame.Angles(wheelSpin, 0, 0)
	for _, wh in ipairs(wheels) do
		local base = wh.out:Lerp(wh.inn, a)
		for _, e in ipairs(wh.parts) do
			if e.w and e.w.Parent then
				if e.spin then e.w.C0 = base * sp * e.off else e.w.C0 = base * e.off end
			end
		end
	end
end

-- Сборка тарелки (v5)
---------------------------------------------------------------------
local function build(cf)
	ship = Instance.new("Model")
	ship.Name = "SaucerX"

	root = Instance.new("Part")
	root.Name = "Root"
	root.Size = Vector3.new(1, 1, 1)
	root.Transparency = 1
	root.Anchored = true
	root.CanCollide = false
	root.CanTouch = false
	root.CanQuery = false
	root.CFrame = cf
	root.Parent = ship
	ship.PrimaryPart = root

	lensList, lightList = {}, {}
	spinners, wheels, pulsers, podFlames = {}, {}, {}, {}
	lastWheelA, lastWheelSpin = -1, -1
	flamePower = 0
	anim = { doors = {}, hood = {}, trunk = {}, gear = {} }
	alpha = { doors = 0, hood = 0, trunk = 0, gear = 0 }
	gen = { doors = gen.doors + 1, hood = gen.hood + 1, trunk = gen.trunk + 1, gear = gen.gear + 1 }
	state.doors, state.hood, state.trunk, state.gear = false, false, false, false

	-- нижняя часть корпуса (сплошная, расширенная до 21.2 стада) и просторный пол салона
	disc("Belly", -1.0, 1.2, 11.4, C.dark, MAT.Metal, { collide = true })
	disc("Lower", -0.15, 0.7, 17.6, C.hull2, MAT.Metal, { collide = true })
	disc("Skirt", 0, 0.4, 21.2, C.hull2, MAT.Metal, { collide = true })
	disc("Seam", 0.12, 0.14, 21.5, C.dark)
	disc("Floor", 0.35, 0.3, 19.8, Color3.fromRGB(58, 62, 62), MAT.DiamondPlate, { collide = true })

	-- нижнее свечение
	local thrust = disc("Thrust", -1.62, 0.1, 6.4, C.orange, MAT.Neon, { tr = 0.25 })
	addLight(thrust, C.orange, 1.2, 18)

	-- кольца стеклянного купола (капсулы)
	local rings = {}
	for i = 0, 4 do
		local ph = math.rad(18 * i)
		rings[i] = { r = DR * math.cos(ph), y = Y2 + DH * math.sin(ph) }
	end
	-- СТЕКЛЯННАЯ КАПСУЛА ПОЛНОСТЬЮ ПРОХОДИМАЯ НАСКВОЗЬ (CanCollide = false, CanQuery = false)
	local noPhys = { tr = 0.64, query = false, collide = false }

	-- стенки корпуса и палуба вокруг купола
	for j = 0, N - 1 do
		local th = thetaOf(j)
		if not DOOR_J[j] then
			local cfA, lenA = slabCF(th, Y0, rOf(Y0), Y1, rOf(Y1))
			mk("WallA", BLOCK, Vector3.new(wallW(rOf(Y0)), lenA, 0.3), cfA, C.hull2, MAT.Metal, { collide = true })
		end
		if not (DOOR_J[j] or HOOD_J[j] or TRUNK_J[j]) then
			local cfB, lenB = slabCF(th, Y1, rOf(Y1), Y2, R2)
			mk("WallB", BLOCK, Vector3.new(wallW(rOf(Y1)), lenB, 0.3), cfB, C.hull, MAT.Metal, { collide = true })
		end
		-- палуба вокруг купола (над дверями она открывается вместе с дверью)
		if not (HOOD_J[j] or TRUNK_J[j] or DOOR_J[j]) then
			mk("Deck", BLOCK, Vector3.new(wallW(R2), 0.3, R2 - DR),
				CFrame.Angles(0, th, 0) * CFrame.new(0, Y2 - 0.15 + (j % 2) * 0.01, -(R2 + DR) / 2), C.hull, MAT.Metal, { collide = true })
			-- обод у основания купола (нефизический, чтобы капсулу можно было проходить насквозь)
			mk("DomeRim", BLOCK, Vector3.new(wallW(DR + 0.05), 0.34, 0.45),
				CFrame.Angles(0, th, 0) * CFrame.new(0, Y2 - 0.02, -(DR + 0.05)), C.dark, MAT.Metal, { query = false, collide = false })
		end
	end

	-- КУПОЛ (СТЕКЛЯННАЯ КАПСУЛА): полностью нефизический — сквозь него можно свободно проходить!
	-- Боковые секции купола над дверями (j = 3, 4 и j = 11, 12 на ярусах i = 0..2) прикреплены к дверям,
	-- чтобы открываться вбок вместе с дверью, точь-в-точь как на фото из Рика и Морти!
	for i = 0, 3 do
		local a, b = rings[i], rings[i + 1]
		for j = 0, N - 1 do
			local isDoorWindow = (i < 3) and DOOR_J[j]
			if not isDoorWindow then
				local pcf, plen = slabCF(thetaOf(j), a.y, a.r, b.y, b.r)
				mk("Dome", BLOCK, Vector3.new(2 * ((a.r + b.r) / 2) * TANH + 0.04, plen, 0.09), pcf, C.glass, MAT.Glass, noPhys)
			end

		-- Округлые валики по краям подушки
		for _, cy in ipairs({ -0.58, 0.58 }) do
			addDoor("DoorCushionRollH", CYL, Vector3.new(3.2, 0.38, 0.38),
				cushCF * CFrame.new(0, cy, 0.08) * CFrame.Angles(0, 0, math.pi / 2), C.cushion, MAT.Fabric, { query = false })
		end
		for _, cx in ipairs({ -1.52, 1.52 }) do
			addDoor("DoorCushionRollV", CYL, Vector3.new(1.25, 0.38, 0.38),
				cushCF * CFrame.new(cx, 0, 0.08), C.cushion, MAT.Fabric, { query = false })
		end
		-- 4 втянутые пуговицы на коричневой подушке двери
		for _, bx in ipairs({ -0.68, 0.68 }) do
			for _, by in ipairs({ -0.28, 0.28 }) do
				addDoor("DoorCushionBtn", BALL, Vector3.new(0.22, 0.22, 0.18),
					cushCF * CFrame.new(bx, by, 0.28), C.cushionBt, MAT.Fabric, { query = false })
			end
		end

		-- Удобная выдвижная ступенька-порог снизу проёма (чтобы легко заходить пешком с земли)
		local T, B = Vector2.new(9.6, 0.45), Vector2.new(13.6, -2.0)
		local rlen = (T - B).Magnitude
		local rang = math.atan2(T.Y - B.Y, T.X - B.X)
		local openCF = CFrame.new(s.sx * (T.X + B.X) / 2, (T.Y + B.Y) / 2 - 0.15, 0.4) * CFrame.Angles(0, 0, -s.sx * rang)
		local closedCF = CFrame.new(s.sx * 6.2, -0.18, 0.4)
		local function ramp(name, size, off, color, mat, solid)
			local rel = closedCF * off
			local p, w = mk(name, BLOCK, size, rel, color, mat)
			table.insert(anim.doors, {
				w = w,
				part = solid and p or nil,
				coll = solid and function(a) return a > 0.5 end or nil,
				fn = function(a) return closedCF:Lerp(openCF, a) * off end,
			})
			return p
		end
		ramp("Ramp", Vector3.new(rlen, 0.26, 4.2), CFrame.new(), Color3.fromRGB(70, 74, 74), MAT.DiamondPlate, true)
		ramp("RampEdge", Vector3.new(rlen, 0.08, 0.14), CFrame.new(0, 0.15, 2.05), C.orange, MAT.Neon)
		ramp("RampEdge", Vector3.new(rlen, 0.08, 0.14), CFrame.new(0, 0.15, -2.05), C.orange, MAT.Neon)
	end

	-- КАПОТ (спереди) и БАГАЖНИК (сзади)
	compartment("hood", -1)
	compartment("trunk", 1)

	ship.Parent = Workspace
	setLights(state.lights)
end

local function skinColor(char)
	for _, n in ipairs({ "LeftHand", "LeftLowerArm", "Left Arm", "RightHand", "Right Arm" }) do
		local p = char:FindFirstChild(n)
		if p and p:IsA("BasePart") then return p.Color end
	end
	return Color3.fromRGB(234, 184, 146)
end

local function buildArms(char)
	if fpFolder then fpFolder:Destroy() end
	fpFolder = Instance.new("Folder")
	fpFolder.Name = "SaucerXArms"
	fpFolder.Parent = Workspace
	local skin = skinColor(char)
	local sleeve = skin
	if char:FindFirstChildOfClass("Shirt") then sleeve = Color3.fromRGB(72, 84, 104) end
	fpArms = {}
	for i, side in ipairs({ -1, 1 }) do
		fpArms[i] = {
			side = side,
			shoulder = (side < 0) and SHOULDER_L or SHOULDER_R,
			phi = (side < 0) and (math.pi - 0.2) or 0.2,
			upper = newFP("Upper", BLOCK, sleeve),
			fore = newFP("Fore", BLOCK, skin),
			elbow = newFP("Elbow", BALL, sleeve),
			shoulderB = newFP("Shoulder", BALL, sleeve),
			palm = newFP("Palm", BLOCK, skin),
			fingers = newFP("Fingers", BLOCK, skin),
			thumb = newFP("Thumb", BLOCK, skin),
		}
	end
end

local function segment(part, a, b, thick)
	local d = b - a
	local len = math.max(d.Magnitude, 0.05)
	part.Size = Vector3.new(thick, thick, len)
	part.CFrame = CFrame.lookAt(a, a + (d.Magnitude > 1e-3 and d or Vector3.new(0, 0, -1))) * CFrame.new(0, 0, -len / 2)
end

-- двухзвенная IK: плечо -> локоть -> запястье
local function solveArm(shoulder, target, l1, l2, pole)
	local d = target - shoulder
	local dist = math.clamp(d.Magnitude, 0.3, (l1 + l2) * 0.995)
	local dir = d.Unit
	local a = (l1 * l1 - l2 * l2 + dist * dist) / (2 * dist)
	local h = math.sqrt(math.max(l1 * l1 - a * a, 0))
	local side = pole - dir * pole:Dot(dir)
	if side.Magnitude < 1e-3 then side = Vector3.new(0, -1, 0) end
	local elbow = shoulder + dir * a + side.Unit * h
	local wrist = shoulder + dir * dist
	return elbow, wrist
end

local function hideCharacter(char, v)
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("BasePart") then d.LocalTransparencyModifier = v end
	end
end

local function updateFP()
	local cam = Workspace.CurrentCamera
	local char = lp.Character
	if not (cam and char and root and fpFolder) then return end
	local base = root.CFrame

	cam.CameraType = Enum.CameraType.Scriptable
	cam.FieldOfView = 80
	cam.CFrame = base * CFrame.new(EYE) * CFrame.Angles(0, fpYaw, 0) * CFrame.Angles(fpPitch, 0, 0)
	pcall(function() UIS.MouseBehavior = Enum.MouseBehavior.LockCenter end)
	hideCharacter(char, 1)

	local W = base * WHEEL_CF
	for _, arm in ipairs(fpArms) do
		local shoulder = (base * CFrame.new(arm.shoulder)).Position
		local rim = W * CFrame.Angles(0, 0, steerAng + arm.phi) * CFrame.new(WHEEL_R, 0, 0)
		local grip = (rim * CFrame.new(0, 0, 0.26)).Position
		local pole = base:VectorToWorldSpace(Vector3.new(arm.side * 0.9, -1, 0.5))
		local elbow, wrist = solveArm(shoulder, grip, UPPER, FORE, pole)

		segment(arm.upper, shoulder, elbow, 0.52)
		segment(arm.fore, elbow, wrist, 0.42)
		arm.elbow.Size = Vector3.new(0.5, 0.5, 0.5); arm.elbow.Position = elbow
		arm.shoulderB.Size = Vector3.new(0.56, 0.56, 0.56); arm.shoulderB.Position = shoulder

		arm.palm.Size = Vector3.new(0.42, 0.34, 0.2)
		arm.palm.CFrame = rim * CFrame.new(0, 0, 0.2)
		arm.fingers.Size = Vector3.new(0.14, 0.36, 0.34)
		arm.fingers.CFrame = rim * CFrame.new(0.17, 0, 0.08)
		arm.thumb.Size = Vector3.new(0.12, 0.14, 0.26)
		arm.thumb.CFrame = rim * CFrame.new(-0.18, 0.08 * arm.side, 0.14) * CFrame.Angles(0, 0, 0.3 * arm.side)
	end
end

function setFirstPerson(on)
	if on then
		if not state.seated then
			say("Сначала сядь за руль (G), потом V")
			return
		end
		local char = lp.Character
		if not char then return end
		buildArms(char)
		fpYaw, fpPitch = 0, 0
		state.fp = true
		say("От 1-го лица: мышь / палец — осмотреться, R — по центру")
	else
		state.fp = false
		if fpFolder then fpFolder:Destroy() fpFolder = nil end
		fpArms = {}
		local cam = Workspace.CurrentCamera
		local char = lp.Character
		if char then hideCharacter(char, 0) end
		pcall(function() UIS.MouseBehavior = Enum.MouseBehavior.Default end)
		if cam then
			cam.CameraType = Enum.CameraType.Custom
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if hum then cam.CameraSubject = hum end
			cam.FieldOfView = 70
		end
	end
	if refreshUI then refreshUI() end
end

function destroyShip()
	if state.fp then setFirstPerson(false) end
	if state.seated then exitShip() end
	if ship then ship:Destroy() end
	ship, root, beacon = nil, nil, nil
	spinners, wheels, pulsers, podFlames = {}, {}, {}, {}
	state.doors, state.hood, state.trunk, state.gear = false, false, false, false
	if refreshUI then refreshUI() end
end

local function spawnShip()
	destroyShip()
	local char = lp.Character
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not hrp then say("Персонаж не найден") return end
	local look = hrp.CFrame.LookVector
	look = Vector3.new(look.X, 0, look.Z)
	if look.Magnitude < 0.01 then look = Vector3.new(0, 0, -1) end
	look = look.Unit
	local p = hrp.Position + look * 26
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = { char }
	local hit = Workspace:Raycast(p + Vector3.new(0, 60, 0), Vector3.new(0, -300, 0), rp)
	local y = hit and (hit.Position.Y + 2.1) or (hrp.Position.Y - 1.0)
	yaw = math.atan2(-look.X, -look.Z) + math.pi
	pitch, roll, vel = 0, 0, Vector3.zero
	pos = Vector3.new(p.X, y, p.Z)
	steerAng, wheelSpin = 0, 0
	build(CFrame.new(pos) * CFrame.Angles(0, yaw, 0))
	refreshUI()
	say("Готово! Z — двери (как на фото), зайди в салон или сквозь капсулу, G — за руль")
end

local function sitDown(hum, hrp)
	yaw = select(2, root.CFrame:ToEulerAnglesYXZ())
	pitch, roll, vel = 0, 0, Vector3.zero
	pos = root.Position
	state.seated = true
	hum.AutoRotate = false
	hrp.Anchored = true
	hrp.CFrame = root.CFrame * seatRel
	hum.Sit = true
	if fly then fly.Visible = true end
	refreshUI()
	task.delay(0.8, function()
		if state.seated and ship then setGroup("doors", false) end
	end)
	say("Взлетай (Space) — из капсул пойдёт синий огонь! V — от 1-го лица")
end

function enterShip()
	if state.seated or busy then return end
	if not ship then say("Сначала создай тарелку") return end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp) then return end
	if (hrp.Position - root.Position).Magnitude > 65 then
		say("Подойди ближе (до 65 стадов)")
		return
	end

	local seatPos = (root.CFrame * seatRel).Position
	local d = hrp.Position - seatPos
	local inside = Vector3.new(d.X, 0, d.Z).Magnitude < 8.5 and math.abs(d.Y) < 7.5
	if inside then
		sitDown(hum, hrp)
		return
	end

	if not state.doors then
		-- снаружи: открываем боковые двери (как на фото)
		setGroup("doors", true)
		say("Двери открыты! Заходи внутрь (или сквозь купол) и нажми G")
		return
	end

	-- двери уже открыты, а игрок ещё снаружи — автопосадка
	busy = true
	task.spawn(function()
		if ship and ship.Parent and hum.Parent then sitDown(hum, hrp) end
		busy = false
	end)
end

function exitShip()
	if not state.seated then return end
	if state.fp then setFirstPerson(false) end
	state.seated = false
	if fly then fly.Visible = false end
	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if hrp and root then
		hrp.Anchored = false
		hrp.CFrame = root.CFrame * CFrame.new(-13.8, 1.2, 1.2)
		hrp.AssemblyLinearVelocity = Vector3.zero
	elseif hrp then
		hrp.Anchored = false
	end
	if hum then
		hum.Sit = false
		hum.AutoRotate = true
		hum:ChangeState(Enum.HumanoidStateType.GettingUp)
	end
	if ship then
		setGroup("doors", true)
		task.delay(3.5, function()
			if ship and ship.Parent and not state.seated then setGroup("doors", false) end
		end)
	end
	refreshUI()
end

local controls
pcall(function()
	local ps = lp:FindFirstChild("PlayerScripts")
	local pm = ps and ps:FindFirstChild("PlayerModule")
	if pm then controls = require(pm):GetControls() end

local function getMove()
	local mv = Vector3.zero
	if controls then
		local ok, v = pcall(function() return controls:GetMoveVector() end)
		if ok and v then mv = v end
	end
	if mv.Magnitude < 0.05 then
		local x, z = 0, 0
		if UIS:IsKeyDown(Enum.KeyCode.W) then z = z - 1 end
		if UIS:IsKeyDown(Enum.KeyCode.S) then z = z + 1 end
		if UIS:IsKeyDown(Enum.KeyCode.D) then x = x + 1 end
		if UIS:IsKeyDown(Enum.KeyCode.A) then x = x - 1 end
		mv = Vector3.new(x, 0, z)
	end
	return mv
end

table.insert(connections, RunService.RenderStepped:Connect(function(dt)
	if not (ship and ship.Parent and root) then return end
	local t = os.clock()
	if beacon then beacon.Transparency = (math.sin(t * 5) > 0) and 0 or 0.7 end

	-- механизмы под капотом, пульсация ядер
	fanAng = fanAng + dt * (state.hood and 16 or 3)
	updateSpinners()
	updatePulses(t)
	updateWheels()

	if not state.seated then
		updatePodFlames(t, dt, state.forceFire, 1, state.forceFire and 0.8 or 0)
		return
	end

	local char = lp.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local hrp = char and char:FindFirstChild("HumanoidRootPart")
	if not (hum and hrp) or hum.Health <= 0 then exitShip() return end

	local mv = getMove()
	local fwd = -mv.Z
	local turn = mv.X
	local boost = UIS:IsKeyDown(Enum.KeyCode.LeftShift) and 2 or 1
	local up = 0
	if UIS:IsKeyDown(Enum.KeyCode.Space) or vertUp then up = up + 1 end
	if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.Q) or vertDown then up = up - 1 end

	-- высота над землёй
	local rp = RaycastParams.new()
	rp.FilterType = Enum.RaycastFilterType.Exclude
	rp.FilterDescendantsInstances = { ship, char, fpFolder }
	local hit = Workspace:Raycast(pos + Vector3.new(0, 3, 0), Vector3.new(0, -600, 0), rp)
	local groundY = hit and hit.Position.Y or -1e5
	local clearance = hit and (pos.Y - 2.1 - groundY) or 999
	local onGround = hit ~= nil and clearance < 0.55 and up <= 0

	local look = CFrame.Angles(0, yaw, 0).LookVector
	local fs = vel:Dot(look)
	if onGround then
		local gain = math.clamp(math.abs(fs) / 10, 0.12, 1)
		yaw = yaw - turn * dt * 1.8 * gain * ((fs < -0.5) and -1 or 1)
	else
		yaw = yaw - turn * dt * 1.8
	end
	local heading = CFrame.Angles(0, yaw, 0)
	look = heading.LookVector

	local target = look * fwd * speed * boost + Vector3.new(0, up * speed * 0.6, 0)
	vel = vel:Lerp(target, math.clamp(dt * 2.5, 0, 1))
	pos = pos + vel * dt

	if hit then
		local floorY = groundY + 2.1
		if onGround then
			pos = Vector3.new(pos.X, pos.Y + (floorY - pos.Y) * math.min(dt * 10, 1), pos.Z)
			vel = Vector3.new(vel.X, 0, vel.Z)
		elseif pos.Y < floorY then
			pos = Vector3.new(pos.X, floorY, pos.Z)
			if vel.Y < 0 then vel = Vector3.new(vel.X, 0, vel.Z) end
		end
	end

	-- колёса: убираются при взлёте, выезжают перед посадкой
	local wantUp = state.gear
	if not hit or clearance > 7 then
		wantUp = true
	elseif clearance < 3.5 + math.max(0, -vel.Y) * 0.7 then
		wantUp = false
	end
	if wantUp ~= state.gear then setGroup("gear", wantUp) end
	if onGround then wheelSpin = wheelSpin - vel:Dot(look) / 1.1 * dt end

	-- СИНИЙ ОГОНЬ ИЗ КАПСУЛ: горит всегда в полёте (и при ускорении/движении)
	local isFlying = (not onGround) or (up > 0) or (math.abs(fwd) > 0.1 and boost > 1) or state.forceFire
	local spdRatio = math.clamp(vel.Magnitude / math.max(speed, 1), 0, 1.5)
	updatePodFlames(t, dt, isFlying, boost, spdRatio)

	-- руль плавно поворачивается
	steerAng = steerAng + (-turn * 1.3 - steerAng) * math.min(dt * 8, 1)

	local tp = onGround and 0 or (-math.clamp(fwd, -1, 1) * 0.14)
	local tr = -math.clamp(turn, -1, 1) * (onGround and 0.05 or 0.3)
	local k = math.min(dt * 4, 1)
	pitch = pitch + (tp - pitch) * k
	roll = roll + (tr - roll) * k

	local bob = onGround and 0 or (math.sin(t * 2) * 0.12)
	root.CFrame = CFrame.new(pos + Vector3.new(0, bob, 0)) * heading * CFrame.Angles(pitch, 0, roll)

	hrp.CFrame = root.CFrame * seatRel
	hrp.AssemblyLinearVelocity = Vector3.zero
	if not hum.Sit then hum.Sit = true end

	if state.fp then updateFP() end
end))

table.insert(connections, UIS.InputBegan:Connect(function(input, gp)
	if gp then return end
	local k = input.KeyCode
	if k == Enum.KeyCode.G then
		if state.seated then exitShip() else enterShip() end
	elseif k == Enum.KeyCode.Z and ship then setGroup("doors", not state.doors)
	elseif k == Enum.KeyCode.X and ship then setGroup("hood", not state.hood)
	elseif k == Enum.KeyCode.C and ship then setGroup("trunk", not state.trunk)
	elseif k == Enum.KeyCode.L then setLights(not state.lights)
	elseif k == Enum.KeyCode.V then setFirstPerson(not state.fp)
	elseif k == Enum.KeyCode.R then fpYaw, fpPitch = 0, 0
	elseif k == Enum.KeyCode.B and ship and not state.seated then setGroup("gear", not state.gear)
	elseif k == Enum.KeyCode.F then
		state.forceFire = not state.forceFire
		refreshUI()
	end
end))

-- осмотр в режиме от 1-го лица (мышь и палец)
table.insert(connections, UIS.InputChanged:Connect(function(input, gp)
	if not (state.seated and state.fp) then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement then
		fpYaw = math.clamp(fpYaw - input.Delta.X * 0.004, -2.4, 2.4)
		fpPitch = math.clamp(fpPitch - input.Delta.Y * 0.004, -1.2, 1.2)
	elseif input.UserInputType == Enum.UserInputType.Touch and not gp then
		fpYaw = math.clamp(fpYaw - input.Delta.X * 0.007, -2.4, 2.4)
		fpPitch = math.clamp(fpPitch - input.Delta.Y * 0.007, -1.2, 1.2)
	end
end))

env.__SaucerX = function()
	for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
	pcall(function() if state.fp then setFirstPerson(false) end end)
	pcall(destroyShip)
	if gui then gui:Destroy() end
end
