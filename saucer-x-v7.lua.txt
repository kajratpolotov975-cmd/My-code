  ЧТО НОВОГО В v7:
  - КУПОЛ ТЕПЕРЬ ТВЁРДЫЙ: стекло собрано из настоящих панелей (CanCollide = true).
    Купол внутри пустой, сквозь стекло пройти нельзя, только через открытую дверь.

    Тап по двери / капоту / багажнику ... открыть / закрыть
    Тап по креслу ....................... сесть (слева пилот, справа пассажир)
    Тап по фаре ......................... свет вкл / выкл
    Джойстик / WASD ..................... лететь (вперёд = туда, куда смотрит камера)
    Прыжок / Space ...................... вверх     Ctrl / Q ... вниз     Shift ... ускорение
    L свет | V вид из кабины | B колёса | G сесть / выйти

  СТОЛКНОВЕНИЯ (v7.1)
    Тарелка больше не пролетает сквозь стены и препятствия: она упирается, скользит вдоль стены

    В коде есть CRASH_DAMAGE: поставь число (например 30), и сильный удар будет ранить персонажа.
  Синее свечение (пламя, голубой свет и неон) убрано.

local function groundY(from, ignore)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = ignore
	local res = Workspace:Raycast(from, Vector3.new(0, -800, 0), params)
	return res and res.Position.Y or nil
end

local groupXf = {}
local function ease(t) return t * t * (3 - 2 * t) end

local function applyAlpha(group, a)
	local xf = groupXf[group]
	local isDoor = (group == "doorL" or group == "doorR")
	for _, e in ipairs(anim[group]) do
		local w = e.w
		if w and w.Parent then
			if e.fn then w.C0 = e.fn(a) else w.C0 = xf(a) * e.closed end
			if e.cc then
				if isDoor then
					-- пока дверь движется, не толкаем игрока; закрытая и открытая (трап) — твёрдые
					e.part.CanCollide = (a <= 0.001 or a >= 0.999)
				else
					e.part.CanCollide = (a <= 0.001)
				end
			end
		end
	end
end

local function applySteer(a)
	for _, e in ipairs(anim.steer) do
		if e.w and e.w.Parent then e.w.C0 = groupXf.steer(a) * e.closed end
	end
end

setGroup = function(group, open, instant)
	state[group] = open
	gen[group] = gen[group] + 1
	local my = gen[group]
	local from, to = alpha[group], open and 1 or 0
	if instant or from == to then
		alpha[group] = to
		applyAlpha(group, to)
		return
	end
	local dur = 0.9
	if group == "hood" or group == "trunk" then dur = 0.7 end
	if group == "gear" then dur = 0.8 end
	task.spawn(function()
		local t0 = os.clock()
		while gen[group] == my and ship do
			local k = math.clamp((os.clock() - t0) / dur, 0, 1)
			alpha[group] = from + (to - from) * ease(k)
			applyAlpha(group, alpha[group])
			if k >= 1 then break end
			RunService.Heartbeat:Wait()
		end
	end)
end

local DA, DB, DC, DE = 4.5, 8.0, 6.5, 0.75   -- полуоси купола и «квадратность» формы

local function spow(v, e)
	if v < 0 then return -((-v) ^ e) end
	return v ^ e
end

-- точка на поверхности купола (th — угол вокруг, ph — от вершины)
local function P(th, ph)
	local s = math.sin(ph)
	return Vector3.new(DA * s * spow(math.cos(th), DE), DB * math.cos(ph), DC * s * spow(math.sin(th), DE))
end

local function mk(name, size, color, mat, transp, shape)
	local p = Instance.new("Part")
	p.Name = name
	if shape then p.Shape = shape end
	p.Size = size
	p.Color = color
	p.Material = mat or Enum.Material.SmoothPlastic
	p.Transparency = transp or 0
	p.Anchored = false
	p.CanCollide = false
	p.CanTouch = false
	p.Massless = true
	p.CastShadow = false
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	return p
end

local function buildShip(spawnCF)

	root.Size = Vector3.new(1, 1, 1)

	root.Anchored = true

	-- добавить деталь: cf — положение относительно центра тарелки
	-- o.group = группа анимации, o.collide = твёрдая, o.query = false (нажатия и лучи проходят),
	-- o.act = действие по нажатию, o.pass = нажатие проходит сквозь деталь, o.fn = своя функция анимации
	local function add(p, cf, o)
		o = o or {}
		p.CFrame = root.CFrame * cf
		p.CanCollide = (o.collide == true)
		p.CanQuery = (o.query ~= false)
		local w = Instance.new("Weld")
		w.Part0 = root
		w.Part1 = p
		w.C0 = cf
		w.C1 = CFrame.new()
		w.Parent = p
		p.Parent = ship
		if o.group then
			anim[o.group][#anim[o.group] + 1] = { w = w, part = p, closed = cf, cc = (o.collide == true), fn = o.fn }
		end
		if o.act then clickActs[p] = o.act end
		if o.pass then passThrough[p] = true end
		return p
	end

	local function B(name, sx, sy, sz, x, y, z, color, mat, o)
		return add(mk(name, Vector3.new(sx, sy, sz), color, mat), CFrame.new(x, y, z), o)

		end

			for j = 0, 2 do
				local p00 = P(th(i), phis[j + 1])
				local p10 = P(th(i + 1), phis[j + 1])
				local p01 = P(th(i), phis[j + 2])
				local p11 = P(th(i + 1), phis[j + 2])
				local center = (p00 + p10 + p01 + p11) / 4
				local w = math.max((p10 - p00).Magnitude, (p11 - p01).Magnitude) * 1.05
				local h = math.max((p01 - p00).Magnitude, (p11 - p10).Magnitude) * 1.08
				local xa = ((p10 - p00) + (p11 - p01)).Unit
				local za = ((p01 - p00) + (p11 - p10)).Unit
				local ya = xa:Cross(za).Unit
				local cf = CFrame.fromMatrix(center, xa, ya)
				local g = doorOf(i, j)
				local panel = mk("Glass", Vector3.new(w, 0.25, h), C.glass, GLS, 0.55)
				panel.Reflectance = 0.08
				if g then
					add(panel, cf, { group = g, collide = true, act = (g == "doorL") and actL or actR })
				else
					add(panel, cf, { collide = true, pass = true })
				end
			end
		end

		-- крышка купола
		local topY = DB * math.cos(phis[4])
		local rx, rz = DA * math.sin(phis[4]), DC * math.sin(phis[4])
		local cap = mk("GlassCap", Vector3.new(rx * 2.1, 0.25, rz * 2.1), C.glass, GLS, 0.55)
		add(cap, CFrame.new(0, topY, 0), { collide = true, pass = true })

		-- вертикальные рёбра
		for k = 0, N - 1 do
			for j = 0, 2 do
				local g = doorOf(k - 1, j) or doorOf(k, j)
				bar(P(th(k), phis[j + 1]), P(th(k), phis[j + 2]), 0.22, C.hull2, nil, g)
			end
		end
		-- кольца: оранжевая LED-лента у основания, рамка над дверью и у крыши
		for i = 0, N - 1 do
			bar(P(th(i), phis[1]), P(th(i + 1), phis[1]), 0.3, C.orange, NEON, doorOf(i, 0))
			bar(P(th(i), phis[3]), P(th(i + 1), phis[3]), 0.22, C.hull2, nil, doorOf(i, 1))
			bar(P(th(i), phis[4]), P(th(i + 1), phis[4]), 0.22, C.hull2, nil, nil)
		end

		-- ручки дверей (за них тоже можно тапнуть)
		for _, d in ipairs({ { "doorR", 1 }, { "doorL", -1 } }) do
			local g, s = d[1], d[2]
			local hp = P(0, math.rad(62))
			local n = Vector3.new(hp.X / (DA * DA), hp.Y / (DB * DB), 0).Unit
			local hpos = hp + n * 0.38
			local ang = math.atan2(n.Y, n.X)
			local cf
			if s > 0 then
				cf = CFrame.new(hpos) * CFrame.Angles(0, 0, ang)
			else
				cf = CFrame.new(-hpos.X, hpos.Y, hpos.Z) * CFrame.Angles(0, 0, math.pi - ang)
			end
			local handle = add(mk("DoorHandle", Vector3.new(0.4, 0.4, 1.7), C.brass, MET), cf,
				{ group = g, act = (s > 0) and actR or actL })
			local nm = (s > 0) and "правая" or "левая"
			addHint(handle, "group", "🚪 Открыть (" .. nm .. ")", "🚪 Закрыть (" .. nm .. ")", g, Vector3.new(0, 1.4, 0))
		end
	end

	-------------------------------------------------- Салон: приборная панель, кресла, штурвал
	B("Dash", 8.0, 1.4, 1.2, 0, 0.7, -3.0, C.hull2, nil, { collide = true })
	B("Screen", 3.2, 0.05, 0.7, 0, 1.42, -3.0, Color3.fromRGB(34, 52, 60), nil, { query = false })
	B("DashLed", 7.0, 0.1, 0.06, 0, 1.1, -2.38, C.orange, NEON, { query = false })
	B("Column", 0.35, 1.8, 0.35, -SEAT_X, 2.1, -2.2, C.black, MET, { query = false })

	local lamp = B("CabinLamp", 0.6, 0.1, 0.6, 0, 7.3, 0, Color3.fromRGB(255, 240, 210), NEON, { query = false })
	local cl = Instance.new("PointLight")
	cl.Range = 16
	cl.Brightness = 0.9
	cl.Color = Color3.fromRGB(255, 232, 200)
	cl.Parent = lamp

	for _, s in ipairs({ -1, 1 }) do
		local act = (s < 0) and actSeatL or actSeatR
		local x = s * SEAT_X
		B("SeatBase", 2.2, 1.0, 2.4, x, 0.5, 0.8, C.seat, FAB, { collide = true, act = act })
		B("SeatCushion", 2.0, 0.4, 2.2, x, 1.2, 0.8, C.cushion, FAB, { act = act })
		local back = B("SeatBack", 2.2, 2.8, 0.5, x, 2.4, 2.1, C.seat, FAB, { collide = true, act = act })
		B("Headrest", 1.6, 0.9, 0.5, x, 4.1, 2.0, C.cushion, FAB, { act = act })
		if s < 0 then
			addHint(back, "seat", "💺 За руль (пилот)", nil, nil, Vector3.new(0, 3.2, 0))
		else
			addHint(back, "seat", "💺 Пассажир (без полёта)", nil, nil, Vector3.new(0, 3.2, 0))
		end
	end

	-- штурвал (поворачивается при манёврах)
	do
		local H = CFrame.new(-SEAT_X, 3.1, -1.7) * CFrame.Angles(math.rad(-25), 0, 0)
		groupXf.steer = function(a) return H * CFrame.Angles(0, 0, a) * H:Inverse() end
		for i = 0, 11 do
			local t = i / 12 * 2 * math.pi
			local lcf = CFrame.new(math.cos(t) * 0.85, math.sin(t) * 0.85, 0) * CFrame.Angles(0, 0, t + math.pi / 2)
			add(mk("WheelRim", Vector3.new(0.5, 0.26, 0.26), C.black, MET), H * lcf, { group = "steer", query = false })
		end
		add(mk("Spoke", Vector3.new(1.7, 0.14, 0.14), C.black, MET), H, { group = "steer", query = false })
		add(mk("Spoke", Vector3.new(1.7, 0.14, 0.14), C.black, MET), H * CFrame.Angles(0, 0, math.pi / 2), { group = "steer", query = false })
		add(mk("WheelHubC", Vector3.new(0.5, 0.5, 0.2), C.orange, NEON), H, { group = "steer", query = false })
	end

	-------------------------------------------------- Готово
	lastCF = root.CFrame
	ship.Parent = Workspace
	setLights(true)
end

local function insideCabin(r)
	if not (root and root.Parent) then return false end
	local p = root.CFrame:PointToObjectSpace(r.Position)
	return math.abs(p.X) < 4.6 and math.abs(p.Z) < 7 and p.Y > -1 and p.Y < 9
end

enterShip = function(side)
	if not ship then return end
	local c, h, r = getChar()
	if not (h and r) or h.Health <= 0 then return end
	if not state.seated then
		local cc = {}
		for _, p in ipairs(c:GetDescendants()) do
			if p:IsA("BasePart") then cc[p] = p.CanCollide end
		end
		saved = { ws = h.WalkSpeed, jp = h.JumpPower, jh = h.JumpHeight, ar = h.AutoRotate, cc = cc }
	end
	state.seated = true
	seatSide = side
	state.pilot = (side == -1)
	h.WalkSpeed = 0
	h.JumpPower = 0
	h.JumpHeight = 0
	h.AutoRotate = false
	pcall(function() h:SetStateEnabled(Enum.HumanoidStateType.Jumping, false) end)
	h.Sit = true
	vel = Vector3.zero
	-- при посадке двери закрываются
	if state.doorL then setGroup("doorL", false) end
	if state.doorR then setGroup("doorR", false) end
	if state.pilot then
		say("Ты за рулём. Джойстик / WASD — лететь, камера задаёт направление.")
	else
		say("Пассажирское место: ты просто сидишь, лететь отсюда нельзя. Хочешь управлять — тапни на левое кресло.")
	end
	refreshUI()
end

exitShip = function(noTeleport)
	if not state.seated then return end
	state.seated = false
	state.pilot = false
	local c, h, r = getChar()
	if h and saved then
		h.WalkSpeed = saved.ws
		h.JumpPower = saved.jp
		h.JumpHeight = saved.jh
		h.AutoRotate = saved.ar
		h.Sit = false
		pcall(function() h:SetStateEnabled(Enum.HumanoidStateType.Jumping, true) end)
	end
	if saved and saved.cc then
		for p, v in pairs(saved.cc) do
			if p and p.Parent then p.CanCollide = v end
		end
	end
	saved = nil
	shake = 0
	if h then h.CameraOffset = Vector3.zero end
	if r and ship and root and root.Parent and not noTeleport then
		local up = 3.4
		if h then up = math.max(3.4, h.HipHeight + r.Size.Y / 2 + 0.7) end
		r.CFrame = root.CFrame * CFrame.new(0, up, 0.6)
		r.AssemblyLinearVelocity = Vector3.zero
		r.AssemblyAngularVelocity = Vector3.zero
	end
	if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
	if ship and curAlt < 2.5 then
		setGroup("doorL", true)
		setGroup("doorR", true)
		say("Ты вышел. Двери открыты — нажми на дверь, чтобы закрыть.")
	else
		say("Ты встал в салоне. Тарелка в воздухе — двери закрыты.")
	end
	refreshUI()
end

local function tryEnter(side)
	if not ship then say("Сначала создай тарелку (кнопка 🛸)"); return end
	local c, h, r = getChar()
	if not (h and r) or h.Health <= 0 then return end
	if state.seated and seatSide == side then return end
	if not state.seated then
		if (r.Position - root.Position).Magnitude > 45 then
			say("Подойди ближе к тарелке")
			refreshUI()
			return
		end
		if not insideCabin(r) and alpha.doorL < 0.9 and alpha.doorR < 0.9 then
			say("Сначала открой дверь — тапни на неё, потом зайди внутрь.")
			refreshUI()
			return
		end
	end
	enterShip(side)
end

ui.create = function()
	if ship then return end
	local c, h, r = getChar()
	if not r then say("Персонаж не найден"); return end
	local look = r.CFrame.LookVector * Vector3.new(1, 0, 1)
	if look.Magnitude < 0.1 then look = Vector3.new(0, 0, -1) end
	look = look.Unit
	local origin = r.Position + look * 26
	local gy = groundY(Vector3.new(origin.X, r.Position.Y + 30, origin.Z), { c }) or (r.Position.Y - 3)
	pos = Vector3.new(origin.X, gy + REST_DOWN, origin.Z)
	local toPlayer = Vector3.new(r.Position.X - pos.X, 0, r.Position.Z - pos.Z)
	local dirv = (toPlayer.Magnitude > 0.1) and toPlayer.Unit or -look
	yaw = math.atan2(-dirv.X, -dirv.Z)   -- нос тарелки смотрит на игрока
	pitch, roll, vel, yawRate = 0, 0, Vector3.zero, 0
	flamePower, lastFlame = 0, -1
	exclTimer = 0
	state.gear, state.gearAuto = false, true
	ui.collapsed = false
	buildShip(CFrame.new(pos) * CFrame.Angles(0, yaw, 0))
	say("Тарелка готова. Тапни на дверь — откроется. Зайди и тапни на левое кресло (пилот).")
	refreshUI()
end

destroyShip = function()
	if state.seated then exitShip(true) end
	setFirstPerson(false)
	for _, g in ipairs(GROUPS) do
		gen[g] = gen[g] + 1
		alpha[g] = 0
		state[g] = false
	end
	if ship then pcall(function() ship:Destroy() end) end
	ship, root = nil, nil
	clickActs, passThrough, hints = {}, {}, {}
	lightList, lensList, flames = {}, {}, {}
	vel = Vector3.zero
	say("Тарелка убрана. Нажми 🛸 чтобы создать")
	refreshUI()
end

local GROUP_TEXT = {
	doorL = { "Левая дверь открыта", "Левая дверь закрыта" },
	doorR = { "Правая дверь открыта", "Правая дверь закрыта" },
	hood  = { "Капот открыт", "Капот закрыт" },
	trunk = { "Багажник открыт", "Багажник закрыт" },
}

local function pickAct(sp)
	local cam = Workspace.CurrentCamera
	if not cam or not ship then return nil end
	local ray = cam:ScreenPointToRay(sp.X, sp.Y)
	local ex = { lp.Character }
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	for _ = 1, 14 do
		params.FilterDescendantsInstances = ex
		local res = Workspace:Raycast(ray.Origin, ray.Direction * 400, params)
		if not res then return nil end
		local part = res.Instance
		local act = clickActs[part]
		if act then return act end
		if passThrough[part] then
			ex[#ex + 1] = part      -- стекло: нажатие проходит насквозь к двери / креслу
		else
			return nil
		end
	end
	return nil
end

local function handleTap(sp)
	local act = pickAct(sp)
	if not act then return end
	if act.kind == "group" then
		setGroup(act.name, not state[act.name])
		say(GROUP_TEXT[act.name][state[act.name] and 1 or 2])
	elseif act.kind == "seat" then
		tryEnter(act.side)
	elseif act.kind == "lights" then
		setLights(not state.lights)
		say(state.lights and "Фары включены" or "Фары выключены")
	end
	refreshUI()
end

local downs = {}
connect(UIS.InputBegan, function(input, gpe)
	local t = input.UserInputType
	if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
		if not gpe then
			local key = (t == Enum.UserInputType.Touch) and input or "mouse"
			downs[key] = { pos = Vector2.new(input.Position.X, input.Position.Y), t = os.clock() }
		end
		return
	end
	if gpe or t ~= Enum.UserInputType.Keyboard then return end
	local k = input.KeyCode
	if k == Enum.KeyCode.G then
		if ship then ui.seatToggle() end
	elseif k == Enum.KeyCode.L then
		if ship then setLights(not state.lights); refreshUI() end
	elseif k == Enum.KeyCode.V then
		if ship then setFirstPerson(not state.fp); refreshUI() end
	elseif k == Enum.KeyCode.B then
		ui.toggleGear()
	end
end)

connect(UIS.InputEnded, function(input)
	local t = input.UserInputType
	if t ~= Enum.UserInputType.MouseButton1 and t ~= Enum.UserInputType.Touch then return end
	local key = (t == Enum.UserInputType.Touch) and input or "mouse"
	local d = downs[key]
	if not d then return end
	downs[key] = nil
	if not ship then return end
	local p = Vector2.new(input.Position.X, input.Position.Y)
	if (p - d.pos).Magnitude < 16 and (os.clock() - d.t) < 0.6 then
		handleTap(p)
	end
end)

connect(UIS.JumpRequest, function() jumpT = os.clock() end)

-- Столкновения: тарелка упирается в стены и препятствия, а не пролетает сквозь них
---------------------------------------------------------------------
local CRASH_DAMAGE = 0      -- урон персонажу от сильного удара (0 = без урона, например 30 = ощутимо)
local HIT_MARGIN = 0.12

local hasBlockcast = pcall(function() return Workspace.Blockcast end)
local castParams = RaycastParams.new()
castParams.FilterType = Enum.RaycastFilterType.Exclude
castParams.IgnoreWater = true
pcall(function() castParams.RespectCanCollide = true end)
local overlapParams = OverlapParams.new()
overlapParams.FilterType = Enum.RaycastFilterType.Exclude
pcall(function() overlapParams.RespectCanCollide = true end)

-- форма тарелки для проверки: низкий длинный корпус + узкий купол сверху
local BOXES = {
	{ off = CFrame.new(0, 0.5, 0), size = Vector3.new(9.2, 3, 23) },
	{ off = CFrame.new(0, 5.0, 0), size = Vector3.new(8.6, 6, 12.6) },
}

local function refreshExclude()
	local t = { ship }
	for _, pl in ipairs(Players:GetPlayers()) do
		if pl.Character then t[#t + 1] = pl.Character end
	end
	excl = t
	castParams.FilterDescendantsInstances = t
	overlapParams.FilterDescendantsInstances = t
end

-- протянуть блок по направлению dir, вернуть первое препятствие
local function sweepBox(base, boxOff, size, dir)
	local from = base * boxOff
	if hasBlockcast then
		local ok, res = pcall(function() return Workspace:Blockcast(from, size, dir, castParams) end)
		if ok then return res end
	end
	-- запасной вариант: лучи из углов и середин блока
	local best
	local hx, hy, hz = size.X / 2, size.Y / 2, size.Z / 2
	for _, sx in ipairs({ -1, 0, 1 }) do
		for _, sy in ipairs({ -1, 1 }) do
			for _, sz in ipairs({ -1, 0, 1 }) do
				local o = (from * CFrame.new(sx * hx, sy * hy, sz * hz)).Position
				local r = Workspace:Raycast(o, dir, castParams)
				if r and (not best or r.Distance < best.Distance) then best = r end
			end
		end
	end
	return best
end

-- последствия удара: тряска, толчок корпуса, сообщение, (по желанию) урон
local function onImpact(impact, n, h)
	local floorHit = n.Y > 0.7
	if impact < (floorHit and 25 or 8) then return end
	shake = math.max(shake, math.min(impact / 60, 1))
	local lv = lastCF:VectorToObjectSpace(n)
	local kick = math.min(impact / 70, 1) * 0.25
	pitch = pitch + lv.Z * kick
	roll = roll - lv.X * kick
	if impact > 14 then say("💥 Удар!") end
	if CRASH_DAMAGE > 0 and state.seated and h and impact > 30 then
		h.Health = h.Health - CRASH_DAMAGE * math.min((impact - 30) / 40, 1)
	end
end

-- сдвинуть тарелку; при столкновении остановиться, скользнуть вдоль стены и отскочить
local function moveWithCollision(p, v, dt, yawv, h)
	local d = v * dt
	local rot = CFrame.Angles(0, yawv, 0)
	for _ = 1, 3 do
		if d.Magnitude < 1e-4 then break end
		local base = CFrame.new(p) * rot
		local hit, hitDist
		for _, bx in ipairs(BOXES) do
			local res = sweepBox(base, bx.off, bx.size, d)
			if res and (not hit or res.Distance < hitDist) then hit, hitDist = res, res.Distance end
		end
		if not hit then
			p = p + d
			break
		end
		if hitDist < 0.02 then
			-- если мы уже внутри чего-то, даём выбраться, а не замираем навсегда
			local inside = false
			for _, bx in ipairs(BOXES) do
				local parts = Workspace:GetPartBoundsInBox(base * bx.off, bx.size, overlapParams)
				if #parts > 0 then inside = true; break end
			end
			if inside then
				p = p + d
				break
			end
		end
		local n = hit.Normal
		local dir = d.Unit
		local adv = math.max(hitDist - HIT_MARGIN, 0)
		p = p + dir * adv
		local vn = v:Dot(n)
		if vn < 0 then
			local impact = -vn
			v = v - n * vn                       -- скользим вдоль поверхности
			onImpact(impact, n, h)
			if impact > 12 and n.Y < 0.7 then
				v = v + n * impact * 0.3         -- отскок от стены
			end
		end
		local rem = d - dir * adv
		d = rem - n * rem:Dot(n)
	end
	return p, v
end

local function update(dt)
	if not ship or not root or not root.Parent then return end
	if dt > 0.05 then dt = 0.05 end
	if dt <= 0 then return end

	local c, h, r = getChar()
	if state.seated and (not h or not r or h.Health <= 0) then exitShip(true) end

	local cam = Workspace.CurrentCamera
	-- лететь может только пилот (левое кресло); пассажир просто сидит
	local flying = state.seated and state.pilot
	local target = Vector3.zero
	local active = false

	if flying and cam then
		local typing = UIS:GetFocusedTextBox() ~= nil
		local mv = getMove()
		local fwd, str = -mv.Z, mv.X
		local upIn = 0
		if not typing then
			if UIS:IsKeyDown(Enum.KeyCode.Space) then upIn = upIn + 1 end
			if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.Q) then upIn = upIn - 1 end
		end
		if (os.clock() - jumpT) < 0.25 or (h and h.Jump) then upIn = math.max(upIn, 1) end
		local cl, cr = cam.CFrame.LookVector, cam.CFrame.RightVector
		local wish = cl * fwd + cr * str + Vector3.new(0, upIn, 0)
		if wish.Magnitude > 1 then wish = wish.Unit end
		active = wish.Magnitude > 0.05
		local boost = 1
		if not typing and (UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift)) then boost = 2 end
		target = wish * speed * boost
		local hl = Vector3.new(cl.X, 0, cl.Z)
		if hl.Magnitude > 0.2 then camYaw = math.atan2(-hl.X, -hl.Z) end
	end

	vel = vel:Lerp(target, 1 - math.exp(-dt * (flying and 2.2 or 3)))
	if not flying and vel.Magnitude < 0.02 then vel = Vector3.zero end
	exclTimer = exclTimer - dt
	if exclTimer <= 0 then
		exclTimer = 0.5
		refreshExclude()
	end
	if vel.Magnitude > 0.01 then
		pos, vel = moveWithCollision(pos, vel, dt, yaw, h)
	end

	-- земля под тарелкой
	local gy = groundY(Vector3.new(pos.X, pos.Y + 3, pos.Z), excl)
	local rest = REST_DOWN + (REST_UP - REST_DOWN) * alpha.gear
	local alt = 999
	if gy then
		local minY = gy + rest
		-- смотрим землю ещё спереди, сзади и по бокам: тарелка плавно едет по склонам,
		-- а высокие ступени и стены (выше ~2.5) считаются препятствием
		local fh = Vector3.new(-math.sin(yaw), 0, -math.cos(yaw))
		local rh = Vector3.new(math.cos(yaw), 0, -math.sin(yaw))
		for _, off in ipairs({ fh * 9, -fh * 9, rh * 3.8, -rh * 3.8 }) do
			local g2 = groundY(Vector3.new(pos.X + off.X, pos.Y + 3, pos.Z + off.Z), excl)
			if g2 and (g2 - gy) <= 2.5 and (g2 + rest) > minY then minY = g2 + rest end
		end
		alt = pos.Y - minY
		if alt < 0 then
			pos = Vector3.new(pos.X, minY, pos.Z)
			alt = 0
			if vel.Y < 0 then vel = Vector3.new(vel.X, 0, vel.Z) end
		end
	end
	curAlt = alt

	-- колёса: сами убираются в полёте и выпускаются у земли
	if alt < 0.6 then
		state.gearAuto = true
		if state.gear then setGroup("gear", false) end
	elseif state.gearAuto and flying then
		local want = alt > 7
		if want ~= state.gear then setGroup("gear", want) end
	end

	-- поворот носом в сторону камеры
	local oldYaw = yaw
	if flying and active then
		local d = (camYaw - yaw + math.pi) % (2 * math.pi) - math.pi
		yaw = yaw + d * (1 - math.exp(-dt * 2.6))
	end
	yawRate = (yaw - oldYaw) / dt

	local fwdH = Vector3.new(-math.sin(yaw), 0, -math.cos(yaw))
	local rightH = Vector3.new(math.cos(yaw), 0, -math.sin(yaw))
	local fv = vel:Dot(fwdH) / (speed * 2)
	local sv = vel:Dot(rightH) / (speed * 2)
	local tp = -fv * 0.45 + (vel.Y / (speed * 2)) * 0.3
	local tr = math.clamp(yawRate * 0.15, -0.5, 0.5) - sv * 0.35
	local k = 1 - math.exp(-dt * 4)
	pitch = pitch + (tp - pitch) * k
	roll = roll + (tr - roll) * k

	-- лёгкое покачивание в воздухе
	local bob = 0
	if alt > 0.5 then bob = math.sin(os.clock() * 1.7) * 0.12 * math.min(alt / 3, 1) end

	local cf = CFrame.new(pos + Vector3.new(0, bob, 0)) * CFrame.Angles(0, yaw, 0) * CFrame.Angles(pitch, 0, roll)
	if (cf.Position - lastCF.Position).Magnitude > 0.003
		or (cf.LookVector - lastCF.LookVector).Magnitude > 0.0008
		or (cf.RightVector - lastCF.RightVector).Magnitude > 0.0008 then
		root.CFrame = cf
		lastCF = cf
	end

	-- персонаж сидит в своём кресле и едет вместе с тарелкой
	if state.seated and r then
		r.CFrame = root.CFrame * CFrame.new(seatSide * SEAT_X, SEAT_TOP + SIT_OFF, 0.8)
		r.AssemblyLinearVelocity = Vector3.zero
		r.AssemblyAngularVelocity = Vector3.zero
		if h and not h.Sit then h.Sit = true end
	end

	-- вращение колёс по земле
	if alt < 0.6 and alpha.gear < 0.5 then
		wheelSpin = wheelSpin - vel:Dot(fwdH) * dt / 1.0
	end
	if math.abs(wheelSpin - lastSpin) > 0.05 then
		lastSpin = wheelSpin
		applyAlpha("gear", alpha.gear)
	end

	-- штурвал
	local st = 0
	if flying then st = math.clamp(yawRate * 0.8, -1.2, 1.2) end
	steerAng = steerAng + (st - steerAng) * (1 - math.exp(-dt * 6))
	if math.abs(steerAng - lastSteer) > 0.02 then
		lastSteer = steerAng
		applySteer(steerAng)
	end

	-- тряска камеры после удара
	if h then
		if shake > 0.01 then
			h.CameraOffset = Vector3.new(math.random() - 0.5, math.random() - 0.5, 0) * shake * 1.4
			shake = shake * math.exp(-dt * 5)
			shakeOn = true
		elseif shakeOn then
			h.CameraOffset = Vector3.zero
			shakeOn = false
			shake = 0
		end
	end

	-- HUD
	hudTimer = hudTimer + dt
	if hudTimer > 0.15 then
		hudTimer = 0
		if hudLabel then
			hudLabel.Text = string.format("⚡ %d    ↑ %d", math.floor(vel.Magnitude + 0.5), math.floor(math.min(alt, 9999) + 0.5))
		end
	end
end

-- пока персонаж сидит, он не сталкивается с тарелкой
connect(RunService.Stepped, function()
	if state.seated and saved and saved.cc then
		for p in pairs(saved.cc) do
			if p and p.Parent then p.CanCollide = false end
		end
	end
end)

connect(lp.CharacterAdded, function()
	if state.seated then
		state.seated = false
		state.pilot = false
		saved = nil
		refreshUI()
	end
end)

pcall(function() RunService:UnbindFromRenderStep("SaucerX_Update") end)
RunService:BindToRenderStep("SaucerX_Update", Enum.RenderPriority.Camera.Value - 1, update)

buildGui()
refreshUI()

env.__SaucerX = function()
	pcall(function() RunService:UnbindFromRenderStep("SaucerX_Update") end)
	pcall(destroyShip)
	for _, cn in ipairs(connections) do pcall(function() cn:Disconnect() end) end
	if gui then pcall(function() gui:Destroy() end) end
	env.__SaucerX = nil
end
