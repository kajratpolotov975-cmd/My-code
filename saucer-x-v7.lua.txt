local function newPart(cf, size, color, material, o)
  o = o or {}
  local p = Instance.new("Part")
  p.Anchored = true
  p.Size = size
  p.CFrame = cf
  p.Color = color
  p.Material = material or Enum.Material.Metal
  p.Transparency = o.transparency or 0
  p.Reflectance = o.reflectance or 0
  p.CanCollide = (o.collide ~= false)   -- по умолчанию ТВЁРДЫЙ
  p.CanQuery = true
  p.CanTouch = false
  p.CastShadow = (o.shadow ~= false)
  p.TopSurface = Enum.SurfaceType.Smooth
  p.BottomSurface = Enum.SurfaceType.Smooth
  if o.shape then p.Shape = o.shape end
  p.Parent = model
  return p
end

local function disc(cf, diameter, thick, color, material, o)
  local oo = {}

  oo.shape = Enum.PartType.Cylinder
  return newPart(cf * CFrame.Angles(0, 0, math.rad(90)), Vector3.new(thick, diameter, diameter), color, material, oo)
end

-- ---------- механизмы (двери / капот / багажник) ----------
local function clearOfChar(m)
  local c = LP.Character
  if not c then return true end
  local params = OverlapParams.new()
  params.FilterType = Enum.RaycastFilterType.Include
  params.FilterDescendantsInstances = {c}
  for _, it in ipairs(m.items) do
    if #Workspace:GetPartsInPart(it.part, params) > 0 then return false end
  end
  return true
end

local function applyMech(m, rootCF)
  local rot = (m.axis == "Y") and CFrame.Angles(0, m.cur, 0) or CFrame.Angles(m.cur, 0, 0)
  local base = rootCF * m.hinge * rot
  for _, it in ipairs(m.items) do
    it.part.CFrame = base * it.rel
  end
end

local function toggleMech(m, silent)
  m.open = not m.open
  m.target = m.open and m.openAngle or 0
  m.solid = false
  for _, it in ipairs(m.items) do it.part.CanCollide = false end
  if not silent then toast(m.label .. (m.open and " — открыто" or " — закрыто")) end
end

local function resetMech(m)
  m.cur, m.target, m.open, m.solid, m.dirty = 0, 0, false, true, true
  for _, it in ipairs(m.items) do it.part.CanCollide = it.collide end
end

local function stepMech(m, dt, rootCF)
  if m.dirty then
    m.dirty = false
    applyMech(m, rootCF)
  end
  if m.cur ~= m.target then
    m.cur = m.cur + (m.target - m.cur) * math.min(1, dt * 7)
    if math.abs(m.target - m.cur) < 0.004 then m.cur = m.target end
    applyMech(m, rootCF)
  elseif (not m.open) and (not m.solid) then
    -- закрылось: делаем твёрдым, но только если в проёме никто не стоит
    if clearOfChar(m) then
      m.solid = true
      for _, it in ipairs(m.items) do it.part.CanCollide = it.collide end
    end
  end
end

local function newMech(label, hingeCF, axis, openAngle, parts)
  local m = {label = label, hinge = hingeCF, axis = axis, openAngle = openAngle,
    cur = 0, target = 0, open = false, solid = true, dirty = false, items = {}}
  for _, p in ipairs(parts) do
    table.insert(m.items, {part = p, rel = hingeCF:ToObjectSpace(p.CFrame), collide = p.CanCollide})
    interact[p] = {fn = function() toggleMech(m) end, range = 45}
  end
  table.insert(mechs, m)
  return m
end

local function closeAll()
  local n = 0
  for _, m in ipairs(mechs) do
    if m.open then toggleMech(m, true); n = n + 1 end
  end
  return n
end

local function ramp(sign)
  local H = CFG.LAND_H
  local run = 6.2
  local theta = math.atan2(H, run)
  local len = math.sqrt(H * H + run * run)
  local cx = 13.4 + run / 2
  local center = Vector3.new(sign * (cx - 0.2 * math.sin(theta)), -H / 2 - 0.2 * math.cos(theta), DZ)
  newPart(CFrame.new(center) * CFrame.Angles(0, 0, -sign * theta), Vector3.new(len, 0.4, 5.6), C_HULL, Enum.Material.DiamondPlate)
end
ramp(1)

local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude
pcall(function() rayParams.RespectCanCollide = true end)

local function groundY(p)
  local ignore = {model}
  if LP.Character then table.insert(ignore, LP.Character) end
  rayParams.FilterDescendantsInstances = ignore
  local r = Workspace:Raycast(p + Vector3.new(0, 4, 0), Vector3.new(0, -4000, 0), rayParams)
  return r and r.Position.Y or nil
end

local function applyRoot()
  local cf = CFrame.new(pos) * CFrame.Angles(0, yaw, 0) * CFrame.Angles(tiltP, 0, tiltR)
  if cf ~= lastCF then
    model:PivotTo(cf)
    lastCF = cf
  end
  return cf
end

local function placeNearPlayer()
  local hrp = getHRP()
  if not hrp then return end
  local look = hrp.CFrame.LookVector
  local d = Vector3.new(look.X, 0, look.Z)
  d = (d.Magnitude > 0.01) and d.Unit or Vector3.new(0, 0, -1)
  local target = hrp.Position + d * CFG.SPAWN_DIST
  local gy = groundY(target)
  pos = Vector3.new(target.X, (gy or (hrp.Position.Y - 3)) + CFG.LAND_H, target.Z)
  yaw = math.atan2(-d.Z, d.X)      -- левая сторона (дверь водителя) смотрит на тебя
  tiltP, tiltR, vel = 0, 0, Vector3.zero
  for _, m in ipairs(mechs) do resetMech(m) end
  lastCF = nil
end

local function isInsideCabin()
  local hrp = getHRP()
  if not hrp or not lastCF then return false end
  local rel = lastCF:PointToObjectSpace(hrp.Position)
  local dx, dz = rel.X, rel.Z - DZ
  return math.sqrt(dx * dx + dz * dz) < R - 0.6 and rel.Y > -1 and rel.Y < R
end

-- ===================== САДИМСЯ / ВСТАЁМ =====================
sitOn = function(s)
  local hum, hrp = getHum(), getHRP()
  if sitSeat or not hum or not hrp or hum.Health <= 0 then return end
  local w = Instance.new("Weld")
  w.Name = "SaucerSeatWeld"
  w.Part0 = s.part
  w.Part1 = hrp
  w.C0 = CFrame.new(0, s.part.Size.Y / 2 + 1.05, 0)
  w.Parent = s.part
  hrp.AssemblyLinearVelocity = Vector3.zero
  hum.Sit = true
  sitSeat, sitWeld, sitTime = s, w, os.clock()
end

standUp = function()
  local hum, hrp = getHum(), getHRP()
  local s = sitSeat
  if sitWeld then pcall(function() sitWeld:Destroy() end) end
  sitWeld, sitSeat = nil, nil
  lastExit = os.clock()
  if hum then hum.Sit = false end
  if hum and hrp and hum.Health > 0 and s and s.part and s.part.Parent then
    hrp.CFrame = s.part.CFrame * CFrame.new(0, s.part.Size.Y / 2 + 3.2, -1.2)
    hrp.AssemblyLinearVelocity = Vector3.zero
  end
end

-- тап по креслу = сесть (только если ты уже внутри кабины)
for _, s in ipairs(seats) do
  local function tapSeat()
    if sitSeat or os.clock() - lastExit < 0.3 then return end
    if not isInsideCabin() then
      toast("Сначала зайди внутрь через дверь")
      return
    end
    sitOn(s)
  end
  interact[s.part] = {fn = tapSeat, range = 14}
  interact[s.back] = {fn = tapSeat, range = 14}
end

local function onModeChanged(old, new)
  hud.Visible = (new == "pilot")
  for _, s in ipairs(seats) do s.bb.Enabled = (sitSeat ~= s) end
  if new == "pilot" then
    autoClosed = false
    toast("Ты за рулём. Стик — лететь, ▲ ▼ — вверх и вниз.")
  elseif new == "passenger" then
    toast("Ты пассажир: тут только сидишь. Чтобы лететь, садись в левое кресло с рулём.")
  elseif old ~= "none" then
    upHeld, downHeld = false, false
  end
end

-- ===================== ГЛАВНЫЙ ЦИКЛ =====================
local function step(dt)
  if not model.Parent then return end
  dt = math.min(dt, 0.1)
  local hum, hrp = getHum(), getHRP()

  -- всё ещё сидим?
  if sitSeat then
    local bad = (not hum) or (not hrp) or hum.Health <= 0 or (not sitWeld) or (not sitWeld.Parent) or sitWeld.Part1 ~= hrp
    if not bad then
      if os.clock() - sitTime < 0.5 then
        hum.Sit = true
      elseif not hum.Sit then
        bad = true            -- нажал прыжок — вышел
      end
    end
    if bad then standUp() end
  end

  -- сам садишься: упёрся в кресло, идя к нему
  if not sitSeat and hum and hrp and hum.Health > 0 and (os.clock() - lastExit) > 1.2 then
    local md = hum.MoveDirection
    if md.Magnitude > 0.1 then
      for _, s in ipairs(seats) do
        local rel = s.part.CFrame:PointToObjectSpace(hrp.Position)
        if math.abs(rel.X) < 2.8 and math.abs(rel.Z) < 2.8 and rel.Y > 0.3 and rel.Y < 4.5 then
          local to = s.part.Position - hrp.Position
          to = Vector3.new(to.X, 0, to.Z)
          if to.Magnitude > 0.1 and md.Unit:Dot(to.Unit) > 0.35 then
            sitOn(s)
            break
          end
        end
      end
    end
  end

  local newMode = sitSeat and sitSeat.kind or "none"
  if newMode ~= mode then
    local old = mode
    mode = newMode
    onModeChanged(old, newMode)
  end

  local maxV = CFG.SPEED * speedMul

  if mode == "pilot" and hum then
    -- ЛЕТАЕТ ТОЛЬКО ВОДИТЕЛЬ
    local mv = hum.MoveDirection
    local mag = math.min(mv.Magnitude, 1)
    local typing = UIS:GetFocusedTextBox() ~= nil
    local up = upHeld or ((not typing) and UIS:IsKeyDown(Enum.KeyCode.E))
    local down = downHeld or ((not typing) and UIS:IsKeyDown(Enum.KeyCode.Q))
    local vy = (up and 1 or 0) - (down and 1 or 0)

    local desired = Vector3.zero
    if mag > 0.05 then desired = mv.Unit * mag * maxV end
    desired = desired + Vector3.new(0, vy * CFG.VSPEED * (0.6 + 0.4 * speedMul), 0)

    if (mag > 0.1 or vy ~= 0) and not autoClosed then
      autoClosed = true
      if closeAll() > 0 then toast("Всё закрылось — поехали!") end
    end

    vel = vel:Lerp(desired, 1 - math.exp(-dt * 3.2))
    if mag > 0.15 then
      local target = math.atan2(-mv.X, -mv.Z)
      yaw = yaw + wrapAngle(target - yaw) * (1 - math.exp(-dt * CFG.TURN))
    end
    pos = pos + vel * dt

    local lv = CFrame.Angles(0, yaw, 0):VectorToObjectSpace(vel)
    local tp = math.clamp(lv.Z / maxV * 0.2, -0.3, 0.3)
    local tr = math.clamp(-lv.X / maxV * 0.25, -0.35, 0.35)
    local k = 1 - math.exp(-dt * 5)
    tiltP = tiltP + (tp - tiltP) * k
    tiltR = tiltR + (tr - tiltR) * k

    local gy = groundY(pos)
    if gy then
      local minY = gy + CFG.LAND_H
      if pos.Y < minY then
        pos = Vector3.new(pos.X, minY, pos.Z)
        if vel.Y < 0 then vel = Vector3.new(vel.X, 0, vel.Z) end
      end
    end
  else
    -- водителя нет (или ты пассажир): тормозим и плавно садимся
    vel = vel:Lerp(Vector3.zero, 1 - math.exp(-dt * 4))
    pos = pos + vel * dt
    local k = 1 - math.exp(-dt * 4)
    tiltP = tiltP * (1 - k)
    tiltR = tiltR * (1 - k)
    local gy = groundY(pos)
    if gy then
      local minY = gy + CFG.LAND_H
      if pos.Y > minY + 0.03 then
        local d = pos.Y - minY
        local fall = math.min(d, math.min(14, 2 + d * 0.7) * dt)
        pos = Vector3.new(pos.X, pos.Y - fall, pos.Z)
      elseif pos.Y < minY then
        pos = Vector3.new(pos.X, minY, pos.Z)
      end
    end
  end

  local rootCF = applyRoot()
  for _, m in ipairs(mechs) do stepMech(m, dt, rootCF) end
end

-- ===================== ТАП ПО ДВЕРЯМ / КАПОТУ / БАГАЖНИКУ =====================
local lastTap = 0

local function castTap(ray)
  local ignore = {}
  if LP.Character then table.insert(ignore, LP.Character) end
  local params = RaycastParams.new()
  params.FilterType = Enum.RaycastFilterType.Exclude
  for _ = 1, 10 do
    params.FilterDescendantsInstances = ignore
    local res = Workspace:Raycast(ray.Origin, ray.Direction * 600, params)
    if not res then return false end
    local p = res.Instance
    local it = interact[p]
    if it then
      local hrp = getHRP()
      if hrp and (hrp.Position - res.Position).Magnitude > it.range then
        toast("Подойди ближе")
      else
        it.fn()
      end
      return true
    end
    if glassSet[p] then
      table.insert(ignore, p)      -- тап проходит сквозь стекло (только тап, пройти сквозь него нельзя)
    else
      return false
    end
  end
  return false
end

local function handleTap(p2, viewportFirst)
  if os.clock() - lastTap < 0.25 then return end
  local cam = Workspace.CurrentCamera
  if not cam then return end
  local a, b
  if viewportFirst then
    a, b = cam:ViewportPointToRay(p2.X, p2.Y), cam:ScreenPointToRay(p2.X, p2.Y)
  else
    a, b = cam:ScreenPointToRay(p2.X, p2.Y), cam:ViewportPointToRay(p2.X, p2.Y)
  end
  if castTap(a) or castTap(b) then lastTap = os.clock() end
end

bind(UIS.InputBegan, function(input, gp)
  if gp then return end
  if input.UserInputType == Enum.UserInputType.MouseButton1 then
    handleTap(UIS:GetMouseLocation(), true)
  end
end)
bind(UIS.TouchTapInWorld, function(p, processed)
  if processed then return end
  handleTap(p, false)
end)
