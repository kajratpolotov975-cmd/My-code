local function weld(a, b, c0)
	local w = Instance.new("Weld")
	w.Part0 = a
	w.Part1 = b
	w.C0 = c0
	w.Parent = b
	return w
end

local function part(parent, name, shape, size, cf, color, mat, o)
	local p = Instance.new("Part")
	p.Name = name
	p.Shape = shape
	p.Size = size
	p.Color = color
	p.Material = mat
	p.Anchored = false
	p.CanCollide = false
	p.CanTouch = false
	p.CanQuery = false
	p.Massless = true
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	if o then
		for k, v in pairs(o) do p[k] = v end
	end
	p.Parent = model
	local w = weld(parent, p, cf)
	return p, w
end
local function blk(parent, name, size, cf, color, mat, o)
	return part(parent, name, BLOCK, size, cf, color, mat, o)
end
local function cylY(parent, name, d, h, cf, color, mat, o)
	return part(parent, name, CYL, Vector3.new(h, d, d), cf * CFrame.Angles(0, 0, PI / 2), color, mat, o)
end
local function cylZ(parent, name, d, len, cf, color, mat, o)
	return part(parent, name, CYL, Vector3.new(len, d, d), cf * CFrame.Angles(0, PI / 2, 0), color, mat, o)
end
local function ball(parent, name, d, cf, color, mat, o)
	return part(parent, name, BALL, Vector3.new(d, d, d), cf, color, mat, o)
end

local lastGear = -1
local function updateGear()
	local ga = S.gear
	local y = 0.2 + (-2.4 - 0.2) * ga
	local resize = math.abs(ga - lastGear) > 0.002
	lastGear = ga
	local sc = math.max(ga, 0.05)
	for _, g in ipairs(gears) do
		g.hw.C0 = CFrame.new(g.x, y, g.z) * CFrame.Angles(0, g.front and S.steerVis or 0, 0)
		g.sw.C0 = CFrame.Angles(S.spin, 0, 0)
		if resize then
			for _, it in ipairs(g.list) do
				it.p.Size = it.size * sc
				it.w.C0 = CFrame.new(it.cf.Position * sc) * (it.cf - it.cf.Position)
			end
		end
	end
end

end

local function fwdFlat(yaw)
	return Vector3.new(-math.sin(yaw), 0, -math.cos(yaw))
end
local function sign(x) return x > 0 and 1 or (x < 0 and -1 or 0) end

local function sampleGround(p, yaw)
	local f = fwdFlat(yaw)
	local rt = Vector3.new(math.cos(yaw), 0, -math.sin(yaw))
	local fy = p.Y + 3
	local c = gy(p.X, p.Z, fy)
	if not c then return nil end
	local R = 6
	local pf = gy(p.X + f.X * R, p.Z + f.Z * R, fy) or c
	local pb = gy(p.X - f.X * R, p.Z - f.Z * R, fy) or c
	local pr = gy(p.X + rt.X * R, p.Z + rt.Z * R, fy) or c
	local pl = gy(p.X - rt.X * R, p.Z - rt.Z * R, fy) or c
	local avg = (c * 2 + pf + pb + pr + pl) / 6
	local pitch = math.clamp(math.atan2(pf - pb, 2 * R), -0.6, 0.6)
	local roll = math.clamp(math.atan2(pr - pl, 2 * R), -0.6, 0.6)
	return avg, pitch, roll
end

local function blocked(origin, dir, dist)
	local r = Workspace:Raycast(origin, dir * dist, rp)
	return r and r.Normal.Y < 0.7
end

local function readInput()

	if controls then

		if ok and r then v = r end
	end
	local th, st = -v.Z, v.X
	if th == 0 and st == 0 and not UIS:GetFocusedTextBox() then
		if UIS:IsKeyDown(Enum.KeyCode.W) or UIS:IsKeyDown(Enum.KeyCode.Up) then th += 1 end
		if UIS:IsKeyDown(Enum.KeyCode.S) or UIS:IsKeyDown(Enum.KeyCode.Down) then th -= 1 end
		if UIS:IsKeyDown(Enum.KeyCode.D) or UIS:IsKeyDown(Enum.KeyCode.Right) then st += 1 end
		if UIS:IsKeyDown(Enum.KeyCode.A) or UIS:IsKeyDown(Enum.KeyCode.Left) then st -= 1 end
	end
	local up = 0
	local hum = getHum()
	local focus = UIS:GetFocusedTextBox()
	if (not focus and UIS:IsKeyDown(Enum.KeyCode.Space)) or btnUp or (hum and hum.Jump) then up += 1 end
	if (not focus and (UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.LeftControl))) or btnDown then up -= 1 end
	if hum then hum.Jump = false end
	return math.clamp(th, -1, 1), math.clamp(st, -1, 1), math.clamp(up, -1, 1)
end

local function updateComps(dt)
	for _, c in pairs(comp) do
		if c.a ~= c.t then
			local d = c.t - c.a
			local stp = dt / 0.8
			if math.abs(d) <= stp then c.a = c.t else c.a += stp * (d > 0 and 1 or -1) end
			local e = c.a * c.a * (3 - 2 * c.a)
			c.w.C0 = c.base * c.fn(e)
		end
	end
end

local function step(dt)
	local th, st, up = 0, 0, 0
	if S.seated then th, st, up = readInput() end
	local g

	-- ЗЕМЛЯ
	if not S.air then
		local sp = S.speed
		if th > 0.05 then
			if sp < 0 then sp += 95 * dt else sp += 38 * th * dt end
		elseif th < -0.05 then
			if sp > 0 then sp += 80 * th * dt else sp += 22 * th * dt end
		else
			local fr = 16 * dt
			if math.abs(sp) <= fr then sp = 0 else sp -= sign(sp) * fr end
		end
		if sp > MAXG then sp = math.max(MAXG, sp - 45 * dt) end
		sp = math.max(sp, -26)
		S.speed = sp

		local sf = math.clamp(math.abs(sp) / 14, 0, 1)
		S.yaw -= st * (1.7 / (1 + math.abs(sp) / 55)) * sf * (sp >= 0 and 1 or -1) * dt

		if up > 0 and (S.vtol or sp >= TAKEOFF) then
			S.air = true
			S.vy = S.vtol and 6 or 14
			S.pos += Vector3.new(0, 0.6, 0)
			notify("Взлёт!")
		else
			local f = fwdFlat(S.yaw)
			local np = S.pos + f * sp * dt
			if math.abs(sp) > 0.5 and blocked(S.pos + Vector3.new(0, 0.5, 0), f * sign(sp), 11.5 + math.abs(sp) * dt) then
				S.speed = -sp * 0.2
				np = S.pos
			end
			local gg, pT, rT = sampleGround(np, S.yaw)
			g = gg
			if gg then
				local ty = gg + CLEAR
				S.pos = Vector3.new(np.X, S.pos.Y + (ty - S.pos.Y) * math.min(1, 14 * dt), np.Z)
				S.pitch += (pT - S.pitch) * math.min(1, 9 * dt)
				S.roll += (rT - S.roll) * math.min(1, 9 * dt)
			else
				S.pos = np
				S.air = true
				S.vy = 0
			end
		end
	end

	-- ВОЗДУХ
	if S.air then
		local vel
		if S.vtol then
			local tsp = th * (th > 0 and 95 or 35)
			S.speed += (tsp - S.speed) * math.min(1, 1.2 * dt)
			S.vy += (up * 38 - S.vy) * math.min(1, 2.2 * dt)
			S.pitch += (-th * 0.22 - S.pitch) * math.min(1, 3 * dt)
			S.roll += (-st * 0.5 - S.roll) * math.min(1, 3 * dt)
			S.yaw -= st * 1.5 * dt
			vel = fwdFlat(S.yaw) * S.speed + Vector3.new(0, S.vy, 0)
		else
			local sp = S.speed
			if th > 0.05 then sp += 60 * th * dt elseif th < -0.05 then sp += 70 * th * dt else sp -= 3 * dt end
			sp -= math.sin(S.pitch) * 34 * dt
			sp = math.clamp(sp, 0, MAXA)
			S.speed = sp
			if math.abs(up) > 0.05 then S.pitch += up * 0.85 * dt else S.pitch += (0 - S.pitch) * 0.45 * dt end
			local sink = 0
			if sp < STALL then
				local k = 1 - sp / STALL
				sink = k * k * 55
				S.pitch += (-0.5 - S.pitch) * k * 1.5 * dt
			end
			S.pitch = math.clamp(S.pitch, -1.0, 1.0)
			S.roll += (-st * 0.62 - S.roll) * math.min(1, 3.2 * dt)
			S.yaw -= st * 1.0 * math.clamp(sp / 45, 0.25, 1) * dt
			local cp = math.cos(S.pitch)
			local f3 = Vector3.new(-math.sin(S.yaw) * cp, math.sin(S.pitch), -math.cos(S.yaw) * cp)
			vel = f3 * sp + Vector3.new(0, -sink, 0)
		end
		S.vy = vel.Y
		local np = S.pos + vel * dt
		local spd = vel.Magnitude
		if spd > 1 and blocked(S.pos, vel.Unit, 11.5 + spd * dt) then
			S.speed *= 0.3
			S.vy = 0
			np = S.pos
		end
		local gg = sampleGround(np, S.yaw)
		g = gg
		if gg and np.Y - CLEAR <= gg then
			S.air = false
			S.pos = Vector3.new(np.X, gg + CLEAR, np.Z)
			if not S.vtol then
				S.speed = math.sqrt(vel.X * vel.X + vel.Z * vel.Z)
			end
			notify("Касание! Катимся по земле")
		else
			S.pos = np
		end
		local floor = Workspace.FallenPartsDestroyHeight + 60
		if S.pos.Y < floor then S.pos = Vector3.new(S.pos.X, floor, S.pos.Z) S.vy = 0 end
	end

	-- шасси
	local alt = g and (S.pos.Y - CLEAR - g) or 9999
	local gt = (not S.air or alt < 35) and 1 or 0
	S.gear += math.clamp(gt - S.gear, -0.8 * dt, 0.8 * dt)
	S.steerVis += (-st * 0.45 - S.steerVis) * math.min(1, 8 * dt)
	S.spin -= (S.air and S.speed * 0.15 or S.speed) / 1.6 * dt

	root.CFrame = CFrame.new(S.pos) * CFrame.Angles(0, S.yaw, 0) * CFrame.Angles(S.pitch, 0, 0) * CFrame.Angles(0, 0, S.roll)
	updateGear()
	updateComps(dt)

	-- эффекты
	local t = os.clock()
	if S.lights then
		for i, p in ipairs(rimList) do
			local k = (math.sin(t * 3 + i * 0.35) + 1) / 2
			p.Color = C_GREEN:Lerp(Color3.fromRGB(25, 130, 70), k * 0.7)
		end
	end
	local brake = S.seated and th < -0.05 and S.speed > 1
	for _, l in ipairs(tails) do
		if brake then l.Material = NEON l.Color = C_RED_B
		elseif S.lights then l.Material = NEON l.Color = C_RED
		else l.Material = SMOOTH l.Color = C_RED_OFF end
	end
	beacon.Color = (math.floor(t * 2) % 2 == 0) and C_RED or Color3.fromRGB(80, 10, 10)
	emitter.Rate = S.seated and (S.air and (50 + S.speed / 3) or 10) or 0
	underLight.Brightness = 1.6 + math.sin(t * 6) * 0.4 + (S.air and 1.2 or 0)

	if S.seated then
		hud.Text = string.format("%s  |  СКОРОСТЬ %d  |  ВЫСОТА %d  |  ШАССИ %s",
			S.air and (S.vtol and "ПОЛЁТ VTOL" or "ПОЛЁТ") or "ЗЕМЛЯ",
			math.floor(math.abs(S.speed)),
			math.floor(math.max(alt, 0)),
			S.gear > 0.5 and "ВЫПУЩЕНЫ" or "УБРАНЫ")
		-- держим персонажа сидящим
		local hum = getHum()
		if hum then hum.Sit = true end
	end
end
