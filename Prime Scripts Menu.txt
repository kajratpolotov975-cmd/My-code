-- Update fly movement local function updateFly() if not flying or not rootPart then return end
local camera = workspace.CurrentCamera local moveDir = Vector3.new(0, 0, 0)
-- Get camera direction local camLook = camera.CFrame.LookVector local camRight = camera.CFrame.RightVector
-- Get input local moveForward = 0 local moveRight = 0 local moveUp = 0
if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveForward = 1 end if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveForward = -1 end if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveRight = -1 end if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveRight = 1 end if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveUp = 1 end if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then moveUp = -1 end
-- Calculate movement direction moveDir = (camLook * moveForward + camRight * moveRight + Vector3.new(0, moveUp, 0))
if moveDir.Magnitude > 0 then moveDir = moveDir.Unit * FLY_SPEED end
bodyVelocity.Velocity = moveDir bodyGyro.CFrame = camera.CFrame end

local function updateSnow() for _, data in ipairs(snowParticles) do local snow = data.frame local currentY = snow.Position.Y.Scale local currentX = snow.Position.X.Scale
-- Move down slowly currentY = currentY + data.speed currentX = currentX + data.drift
-- Reset when off screen if currentY > 1.05 then currentY = -0.05 currentX = data.startX end
-- Keep X in bounds if currentX < 0 then currentX = 0 end if currentX > 1 then currentX = 1 end
snow.Position = UDim2.new(currentX, 0, currentY, 0) end end
